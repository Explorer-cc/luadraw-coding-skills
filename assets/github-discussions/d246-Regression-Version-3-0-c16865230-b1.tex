\documentclass[border = 3pt]{standalone}

\usepackage{luadraw}

\begin{document}

\begin{luadraw}{}
local ld = luadraw

local graphview = ld.graph:new{
  window = {-7, 7, -5, 5},
  bbox   = false
}

graphview:Daxes(
  {0, 1, 1},
  {
    arrows = "->",
    grid   = true
  } )  -- < this is allowed!

graphview:Show()
\end{luadraw}

\end{document}
