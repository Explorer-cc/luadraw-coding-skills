local A,B,C,D,E,F,G,H= Origin, M(13,0,0),  M(13,8,0), M(0,8,0), M(3,4,12), M(10,4,12),M(3,4,0),M(10,4,0)

    g:addFacet(cvx_hull3d(L), {color="Orange",edge=true}),
    g:addPolyline({{E,G},{F,H},{A,G,D},{B,H,C},{H,G}})
    )
g:Dlabel3d("$A$",A,{pos="S"}, "$B$",B,{}, "$C$",C,{}, "$D$",D,{},"$E$",E,{pos="N"}, "$F$",F,{}, "$G$",G,{pos="S"}, "$H$",H,{})
