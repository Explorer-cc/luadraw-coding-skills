\documentclass[border=2 mm]{standalone}
\usepackage[svgnames]{xcolor}
\usepackage[3d]{luadraw}
\usepackage{fouriernc}
\begin{document}
\begin{luadraw}{name=stereographic_curve}
local g = graph3d:new{window={-1.5,4.25,-2.5,4.5},size={10,10}, viewdir={10,75}}
local sqrt = math.sqrt
local A,B,C = M(0,0,0), M(0,2,0), M(sqrt(3),1,0) -- not M(1,sqrt(3),0)
local H = (B+C)/2 -- useful ?
local G = (A+B+C)/3
local S = M(G.x,G.y,sqrt(69)/3) -- not M(1/3, (2+sqrt(3))/3, sqrt(69)/3)
local R = 1
local coefA, coefB, coefC = pt3d.dot(A-S,g.Normal), pt3d.dot(B-S,g.Normal), pt3d.dot(C-S,g.Normal)
local paramA = {A,R,{color="Orange", mode=2,opacity=0.8}}  -- paramaters of sphere A
local paramB = {B,R,{color="Blue",mode=2, opacity=0.8}}
local paramC = {C,R,{color="Red",mode=2, opacity=0.8}}
local before, after = {}, {}
if coefA < 0 then table.insert(before, paramA) -- A is behind S
else table.insert(after,paramA)
end
if coefB < 0 then table.insert(before, paramB) -- B is behind S
else table.insert(after,paramB)
end
if coefC < 0 then table.insert(before, paramC) -- C is behind S
else table.insert(after,paramC)
end
for _,s in ipairs(before) do -- before S
    g:Dsphere(table.unpack(s))
end
g:Dsphere(S,2*R,{color="Green",mode=2,opacity=0.5})  
for _,s in ipairs(after) do -- after S
    g:Dsphere(table.unpack(s))
end
g:Ddots3d({A,B,C,G,H,S})
g:Dpolyline3d( {{A,S},{B,S},{S,C},{A,G},{B,G},{G,C},{A,B,C,A}}, "dashed")
g:Dlabel3d("$A$",A,{pos="S"}, "$B$",B,{},"$C$",C,{pos="S"},"$S$",S,{pos="N"},"$G$",G,{pos="N"})
g:Show()
\end{luadraw}
\end{document}
