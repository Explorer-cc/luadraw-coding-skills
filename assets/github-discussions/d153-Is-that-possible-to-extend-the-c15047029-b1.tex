\documentclass[border=5pt]{standalone}
\usepackage[3d]{luadraw}
\usepackage[svgnames]{xcolor}
\begin{document}
\begin{luadraw}{name=mwe}
    local g = graph3d:new{
        window={-4,4,-4,4},
        margin={0,0,0,0},
        viewdir={30,75},
        size={10,10}
    } 
    local sqrt = math.sqrt
    Hiddenlines = true; 
    require 'luadraw_spherical'
    local R = 2*sqrt(3)
    local P = parallelep(M(-2,-2,-2),4*vecI,4*vecJ,4*vecK)
    local Q = parallelep(M(-1.5,-1.5,-1.5),3*vecI,3*vecJ,3*vecK) -- Q inside the sphere, Origin is the center
    g:Define_sphere({radius=R,color="LightGray!5"})
    Hiddenlinestyle = "solid"
    g:DSpolyline( facetedges(Q), {color="black"})
    Hiddenlinestyle = "dashed"
    for _, edge in ipairs(facetedges(Q)) do
        g:DSarc(edge,1) -- The two vertices of the edge are automatically projected onto the sphere from the center.
    end
    for _,S in ipairs(Q.vertices) do
        g:DSseg({Origin,toSphere(S)},{width=1,color="teal"}) -- toSphere(S) is the projection of S onto the sphere from the center 
    end
    g:DSdots(P.vertices)
    g:Dspherical()
    g:Show()
\end{luadraw}
\end{document}
