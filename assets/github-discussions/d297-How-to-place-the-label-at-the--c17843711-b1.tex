\documentclass{standalone}
\usepackage{luadraw}
\usepackage[svgnames]{xcolor}
\usepackage{fourier-otf}
\begin{document}
\begin{luadraw}{name=tilted_hyperbola}
local ld = luadraw
local Z, i = ld.cpx.Z, ld.cpx.I
local sinh, cosh = math.sinh, math.cosh

local g = ld.graph:new{window = {-6, 6, -4, 7}}
require 'luadraw_decorations'
local right_branch = function(t) return Z(2 * cosh(t), sinh(t)) end
local left_branch = function(t) return Z(-2 * cosh(t), sinh(t)) end

local plot_hyperbola = function(opt, angle)
    angle = angle or 0
    if angle ~= 0 then 
        g:Savematrix(); g:Rotate(angle); g:Labelangle(angle)
    end
    g:Dline({0, 1}, {draw_options="-Latex,ultra thick", label="$x$", anchor1d=0, pos="E"})
    g:Dline({0, i}, {draw_options="-Latex,ultra thick", label="$y$", anchor1d=1, pos="N"})
    g:Dparametric(right_branch, {t = {-2.5, 2.5}, nbdots = 100,draw_options = opt})
    g:Dparametric(left_branch, {t = {-2.5, 2.5}, nbdots = 100,draw_options = opt})
    g:Dline({0, Z(1, 1 / 2)}, "dashed,"..opt)
    g:Dline({0, Z(1, -1 / 2)}, "dashed,"..opt)
    local _, xmax, _, ymax = table.unpack(g:Getview())
    g:Dlabel("$O$",0, {pos="SW"})
    if angle ~= 0 then g:Restorematrix(); g:Labelangle(0)  end
end

plot_hyperbola("teal")

g:Savematrix()
g:Shift(Z(-2, 2))
plot_hyperbola("magenta",45)
g:Restorematrix()

g:Show()
\end{luadraw}
\end{document}
