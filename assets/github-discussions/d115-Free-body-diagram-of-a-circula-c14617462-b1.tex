\begin{luadraw}{name=regular_pyramid_02}
local g = graph3d:new{window={-2,2,-1,4},viewdir={70,70},size={10,10}}
g:Linewidth(6); Hiddenlines = true; Hiddenlinestyle = "dashed"
local a, n, h = 3, 3, 4
local O = Origin
local P = regular_pyramid(n,a,h,false,O) -- n,side,height,open,center,axe
local A,B,C,S = table.unpack(P.vertices)
local X=(S+A)/2
local T = M(0,0,-0.3)
local I,r,n = circumcircle3d(A,B,C) -- a second n variable ?
g:Dcylinder(I,T,r,{color="gray"})
g:Writeln("\\tikzset{->-/.style={decoration={markings, mark=at position #1with {\\arrow{Stealth}}},postaction={decorate}}}")
g:Dpolyline3d({{S,A},{S,B},{S,C}},false,"->-=0.5")
g:Ddots3d(P.vertices)
g:Dlabel3d("$A$",A,{pos="S"}, "$B$",B,{}, "$C$",C,{}, "$\\vv{F_1}$",(S+A)/2,{pos="E"}, "$\\vv{F_3}$",(S+C)/2,{pos="E"}, "$\\vv{F_2}$",(S+B)/2,{pos="W"},"$S$",S,{pos="N"})
g:Show()
\end{luadraw}
