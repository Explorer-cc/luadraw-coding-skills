local ld = luadraw

function ld.common_tangent(p1,u1,u2,p2,v1,v2,grid)
    local h = math.min(1e-6, (u2-u1)/10, (v2-v1)/10)
    local det = ld.cpx.det
    local function diff(f,t1,t2,t) 
        if t+h>t2 then return (f(t2)-f(t-h))/(t2-t+h) end
        if t-h<t1 then return (f(t+h)-f(t1))/(t+h-t1) end
        return (f(t+h)-f(t-h))/(2*h)
    end
    local function F(x,y) return det(diff(p2,v1,v2,y), diff(p1,u1,u2,x)) end
    local function G(x,y) return det(p2(y)-p1(x), diff(p1,u1,u2,x)) end
    local L1 = ld.implicit(F,u1,u2,v1,v2, grid)
    local L2 = ld.implicit(G,u1,u2,v1,v2, grid)
    return ld.interL(L1,L2)
end

function ld.graph:Dcommon_tangent(p1, p2, options)
    options = options or {}
    local t1 = options.t1 or {-5,5} -- p1 parameter limits 
    local t2 = options.t2 or {-5,5} -- p2 parameter limits 
    local grid = options.grid or {50,50}
    local draw = options.draw_options or ""
    local scale  = options.scale or 1
    local I = ld.common_tangent(p1,t1[1],t1[2], p2,t2[1],t2[2], grid)
    print("ok")
    if I ~= nil then
        local out = {}
        for _, z in ipairs(I) do
            local a, b = p1(z.re), p2(z.im)
            self:Dline(a, b, draw, scale)
            table.insert(out, {a,b})
        end
        if options.out ~= nil then table.append(options.out, out) end
    end
end
\end{luacode*}

\begin{luadraw}{name=common_tangent}
local ld = luadraw
local cpx = ld.cpx
local Z = cpx.Z

local function param_path( P ) -- parametrization of a path P (in [0;1])
    local f = ld.curvilinear_param( ld.path( P ) )
    return function(t) 
            if t<=0 then return f(0)
            elseif t>=1 then return f(1)
            else return f(t)
            end
        end
end

local g = ld.graph:new{ size={10,10} }
local P1 = {Z(-4.5,3), Z(-2,3), Z(-1,3), 0 , "b", Z(1,-3), Z(2,-3), Z(4,4), "b"} -- path1
local P2 = ld.spline({Z(-4,1), Z(-1,0), Z(-2,-3), Z(-5,-4)}) -- path2

g:Dpath(P1, "thick,red"); g:Dpath(P2, "thick,blue")
local dots = {}
g:Dcommon_tangent(param_path(P1), param_path(P2), {t1={0,1},t2={0,1}, out=dots})
g:Ddots(dots)
g:Show()
\end{luadraw}
\end{document}
