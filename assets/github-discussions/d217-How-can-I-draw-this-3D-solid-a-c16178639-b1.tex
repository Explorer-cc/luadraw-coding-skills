local draw = function(g, S) -- axes and surface
    g:Labelsize("small") 
    g:Setmatrix( {0, g:Proj3d(vecI), g:Proj3d(vecJ)} )
    g:Labeldir( {g:Proj3d(vecI), g:Proj3d(vecJ)})
    g:Daxes(nil, {limits={{-3,2.5},{-2.5,2.5}}, arrows="-stealth", 
        legend={"$x$", "$y$"}, legendpos={1,1}, legendsep={0,0}, legendstyle={"E","N"}})
    g:IDmatrix()
    g:Dfacet(S, {usepalette={palGasFlame,"x"}, opacity=0.8, edgecolor="lightgray", edgewidth=1})
end

function solid1()
    local cos, sin, pi = math.cos,math.sin, math.pi
    local g = graph3d:new{window3d = {-2,2,-2,2,0,4},adjust2d=true,size={12,12}, bbox=false}
    local p1 = function(u,v) return Mc(2,u,4*v*math.abs(sin(u))) end
    local p2 = function(u,v) return M(2*cos(u),2*(1-2*v)*sin(u),4*sin(u)) end
    local S1 = surface(p1,0,2*pi,0,1,{100,2})
    local S2 = surface(p2,0,pi,0,1,{50,2})
    local u = 35*deg
    g:Dpolyline3d({ p1(u,0), p1(-u,0)}, "blue,line width=0.8")
    draw(g, concat(S1,S2) )
    g:Dpolyline3d( {p1(u,0), p1(u,1), p1(-u,1),p1(-u,0)}, "blue,line width=0.8")
    g:Sendtotex()
end

function solid2()
    local cos, sin, pi = math.cos,math.sin, math.pi
    local g = graph3d:new{window3d = {-2,2,-2,2,0,4},adjust2d=true,size={12,12}, bbox=false}
    local p = function(u,v) return M(2*cos(u), 2*(1-v)*sin(u), 2*v*math.sqrt(3)*math.abs(sin(u))) end
    local S = surface(p,0,2*pi,0,1,{100,2})
    local u = 35*deg
    g:Dpolyline3d({ p(u,0), p(-u,0)}, "blue,line width=0.8")
    draw(g, S)
    g:Dpolyline3d( {p(u,0), p(u,1), p(-u,0)}, "blue,line width=0.8")
    g:Sendtotex()
end

function solid3()
    local cos, sin, pi = math.cos,math.sin, math.pi
    local g = graph3d:new{window3d = {-2,2,-2,2,0,4},adjust2d=true,size={12,12}, bbox=false}
    local p = function(u,v) return M(2*cos(u), 2*sin(u)*sin(v), 2*sin(u)*cos(v)) end
    local S = surface(p,0,pi,pi/2,-pi/2,{20,20})
    local u = 35*deg
    g:Dpolyline3d({ p(u,pi/2), p(u,-pi/2)}, "blue,line width=0.8")
    draw(g, S)
    g:Darc3d( p(u,pi/2), 2*cos(u)*vecI, p(u,-pi/2), 2*sin(u), 1, vecI, "blue,line width=0.8")
    g:Sendtotex()
end

function solid4()
    local cos, sin, pi = math.cos,math.sin, math.pi

    local u = 35*deg

end
