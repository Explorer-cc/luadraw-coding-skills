\documentclass[border=5pt]{standalone}
\usepackage[svgnames]{xcolor}
\usepackage[3d]{luadraw}
\begin{document}
\begin{luadraw}{name=Steinmetz_solid}
local pi = math.pi
local g = graph3d:new{window={-2,2,-2,2},size={12,12}}
require 'luadraw_spherical'
g:Define_sphere({radius=2, show=false})
local axe = {Origin, vecK}
local p1 = function(u,v) return (1-v)*Ms(2,-pi/4,u)+v*Ms(2,pi/4,u) end
local S1 = surface(p1,0,pi,0,1,{50,2})
local S = concat(S1, rotate3d(S1,90,axe), rotate3d(S1,180,axe), rotate3d(S1,-90,axe))
g:DSbigcircle({vecI+vecJ, vecK},{color="red", hidden=true})
g:DSbigcircle({vecI-vecJ, vecK},{color="red", hidden=true})
g:Dfacet(S, {usepalette={palGasFlame,"z"}, opacity=0.7, mode=mShadedOnly})
g:Dspherical()
g:Show()
\end{luadraw}
\end{document}
