function chain(a,b,h)
-- a and b are two 2d points (complex numbers)
-- h = height  of wave
-- returns a path
    h = h or 0.0625
    local n = math.floor(cpx.abs(b-a)/(1.5*h)) -- number of zigzag
    local up,down = {a}, {"b"} -- to build path
    local pas = (b-a)/n
    local v = cpx.I*(b-a)
    v = h*v/cpx.abs(v)
    local x1, x2, c, d = a, a+pas
    for k = 1, n do
        c, d = (x1+x2)/2, (x1+x2)/2
        insert(up,{c+v,d-v,x2,"b"})
        down = concat({"b",d+v,c-v,x1},down)
        x1 = x2; x2 = x2+pas
    end
    table.remove(down,1)
    return concat(up,down)
end

function graph:Dchain3d(seg,h,draw_options)
    if type(h) == "string" then draw_options = h; h = 0.0625 end
    if type(seg[1]) ~= "table" then seg = {seg} end
    for _, s in ipairs(seg) do
        local a,b = table.unpack(self:Proj3d(s))
        self:Dpath(chain(a,b,h), draw_options)
    end
end
\end{luacode*}
