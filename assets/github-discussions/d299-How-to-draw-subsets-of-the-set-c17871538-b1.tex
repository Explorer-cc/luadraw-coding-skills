\documentclass[border=5mm]{standalone}
\usepackage[svgnames]{xcolor}
\usepackage{luadraw} 
\usepackage{amsmath,amssymb}
\begin{document}
\begin{luadraw}{name=interval}
local ld = luadraw
local cpx = ld.cpx
local Z = cpx.Z
local g = ld.graph:new{ window = {0,1.8,-1, 2}, size = {12,3,0}, bbox = false }
local wd = 0.25
local inf = math.huge
local function interval(x1, x2, y, left, right, options)
-- options = { legend="", leftlabel=x1, rightlabel=x2, pattern=<default> }
    options = options or {}
    local legend = options.legend or ""
    local labelx1 = options.leftlabel or x1 --default value
    local labelx2 = options.rightlabel or x2 --default value
    local pattern = options.pattern or "pattern=north east lines,pattern color=gray" --default value
    local u1,u2,v1,v2 = table.unpack( g:Getview() )
    local dx = 0.25/g.Xscale -- to see the arrow
    left = "$\\mathbf{"..left.."}$"
    right = "$\\mathbf{"..right.."}$"
    if x1 > x2 then x1, x2 = x2, x1 end
    if x1 == -inf then x1 = u1 end
    if x2 == inf then x2 = u2 end
    if x1 > u1 then
        g:Drectangle(Z(u1,y-wd/2), Z(x1,y+wd/2), "draw=none,"..pattern)
        g:Dlabel(left, Z(x1,y), {}, "$"..labelx1.."$", Z(x1,y), {pos="S",dist=0.1})
    end
    if x2 < u2 then
        g:Drectangle(Z(x2,y-wd/2), Z(u2-dx,y+wd/2), "draw=none,"..pattern)
        g:Dlabel(right, Z(x2,y), {}, "$"..labelx2.."$", Z(x2,y), {pos="S",dist=0.1})
    end
    g:Dlabel(legend, Z((x1+x2)/2,y), {pos="N"})
    g:Dline({Z(x1,y),1}, "thick,-Stealth")
end

interval(1/15, math.sqrt(2), 1, "[", "]",
    {leftlabel="\\frac{1}{15}", rightlabel="\\sqrt{2}", pattern="fill=blue"})
interval(1/7, 1, 1, "(", "]",
    {leftlabel="\\frac{1}{7}", pattern="pattern=north west lines,pattern color=red"})
g:Show()
\end{luadraw}
\end{document}
