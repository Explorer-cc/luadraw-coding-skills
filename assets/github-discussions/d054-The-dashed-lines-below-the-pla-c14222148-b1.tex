local g = graph3d:new{window={-5,5,-2,5.5},viewdir={30,70},size={10,10}}
local i = cpx.I
g:Linejoin("round"); g:Linewidth(6); Hiddenlines = true; Hiddenlinestyle = "dashed"

function dome(O,V,r,n,angle,m) -- build the dome as a list of facets
-- O=center of base, V=axis (vector), r=radius, n=number of sides,
-- optional: angle to rotate around the axis, m=number of points on edges
    m = m or 20
    angle = angle or 0
    local h = pt3d.abs(V)
    local v = pt3d.normalize(V)
    local S = O+V -- top of dome
    local ellipticEdge = function(M)
        local u = pt3d.normalize(M-O)
        local p = function(t)
            return O+r*math.cos(t)*u+h*math.sin(t)*v
        end
        local L = parametric3d(p,0,90*deg,m,false,0)
        return L[1]
    end
    local u1 = pt3d.prod(V,vecJ)
    if pt3d.N1(u1) < 1e-12 then u1 = pt3d.prod(V,vecI) end
    u1 = pt3d.normalize(u1)
    local u2 = pt3d.prod(v,u1)
    local base = map(function(z) return O+z.re*u1+z.im*u2 end,rotate(polyreg(0,r,n),angle))
    local rep = {reverse(base)}  -- list of facets
    local edges = {}  -- list of edges
    for _,M in ipairs(base) do
        table.insert(edges,ellipticEdge(M))
    end
    table.insert(edges,base); table.insert(edges[n+1],base[1]) -- to close the last list
    local B, A = base[1]
    local Lb, La = edges[1]
    for k = 2, n+1 do
        if k == n+1 then k = 1 end -- to close the list
        A = B; B = base[k]; La = Lb; Lb = edges[k]
        for j = 1, m-2 do
            local facet = {La[j],Lb[j],Lb[j+1],La[j+1]}
            table.insert(rep,facet)
        end
        table.insert(rep,{La[m-1],Lb[m-1],Lb[m]})
    end
    return rep, edges
end

local D1, edges1 = dome(M(-2,-4,0),4*vecK,2,4,15)
local D2, edges2 = dome(M(-2,2,0),3*M(1,0,1),2,6)
-- there is no intersection between D1 and D2, we can use two scene3d.
g:Dscene3d(g:addFacet(D1,{color="LightBlue"}), g:addPolyline(edges1))
g:Dscene3d(g:addFacet(D2,{color="Crimson"}), g:addPolyline(edges2))
g:Show()
\end{luadraw}
