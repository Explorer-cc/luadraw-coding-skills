\documentclass{standalone}
\usepackage[3d]{luadraw} % version 3.5
\usepackage[svgnames]{xcolor}
\usepackage{fourier-otf}
%https://tex.stackexchange.com/questions/129979/how-to-draw-spherical-triangles
\begin{document}
\begin{luadraw}{name=spherical_triangles}
require 'luadraw_spherical'
local ld = luadraw
local pt3d = ld.pt3d
local Origin = pt3d.Origin
local g = ld.graph3d:new{ window={-4.5,4.5,-4.5,4.5}, viewdir={30,65}, size={10,10} }
ld.Hiddenlines = true
local Ct, R = Origin, 4

g:Define_sphere{center=Ct, radius=R, edgestyle = "noline"}
local sM = ld.sM
local A,B,C = sM(30,25), sM(-20,80), sM(80,85)

g:DSfacet({A,B,C}, {fill="blue!50",color="blue", fillopacity=0.5, width=10})
g:DSpolyline({{Ct, A}, {Ct, B}, {Ct, C}}, {width=8})
g:DSdots(Ct)
g:DSlabel("$O$",Ct,{pos="NE"})
local options = function(name)
    return {label=name, pos="center", dist=0.5, fill="pink", fillopacity=0.6, width=6} --<- new options for DSangle
end
g:DSangle(B,A,C,1,1, options("$\\alpha$"))
g:DSangle(A,B,C,1,1, options("$\\beta$"))
g:DSangle(A,C,B,1,1, options("$\\gamma$"))
g:Dspherical()
g:Dballdots3d({A,B,C},"black")
g:Dlabel3d(
    "$B$",B,{pos="SW"},
    "$C$",C,{pos="SE"},
    "$A$",A,{pos="N",dist=0.1}
)
g:Show()
\end{luadraw}
\end{document}
