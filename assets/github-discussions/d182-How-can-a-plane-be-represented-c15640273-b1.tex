local frac = function(val, max)  -- converts val to a fraction (string for TeX)
    max = max or 10000 -- to stop if the process is too long
    local sg, den =  "", 1
    if val < 0 then sg = "-"; val = -val end
    local num = val
    while (den <= max) and (math.abs(num-nearest(num))>1e-8) do den = den+1; num = val*den end
    if den > max then 
        return sg..val -- failure
    else
        num, den = simplifyFrac(nearest(num), den)
        if den > 1 then 
            return sg.."\\frac{"..num.."}{"..den.."}"
        else
            return sg..num
        end
    end
end

function graph3d:Dcoord3d(...)
    local args = {}
    local name, dot
    for k, aux in ipairs{...} do
        if k%3 == 1 then
            name = aux
        elseif k%3 == 2 then
            dot = aux
        else
            insert(args,{"$"..name.."("..frac(dot.x)..","..frac(dot.y)..","..frac(dot.z)..")$", dot, aux})
        end
    end
    self:Dlabel3d(table.unpack(args))
end

local P = planeEqn(2,-3,3,-17)
local A,B = M(3,-4,7), M(-5,-14,13)
local H = proj3d(A,P)
local R = 2*H - A
local I = interDP({B, B-R},P)
local x1,x2,y1,y2,z1,z2 = getbounds3d( {A,B,H,R} ) -- 3d bounding box
local dl = math.max(x2-x1,y2-y1,z2-z1) + 1 -- the greatest length, plus one

Hiddenlinestyle = "dashed";Hiddenlines = true;
local C,u,v = orthoframe(P)
local len1, len2 = 6*u, 5*v
C = (I+H)/2
local F = {C+len1+len2, C-len1+len2, C-len1-len2, C+len1-len2}
g:Dscene3d(
    g:addFacet(F, {edge=true}),
    g:addPolyline({{A,R,B}})

g:Show()
