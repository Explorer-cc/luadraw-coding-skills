\documentclass[border=5pt]{standalone}
\usepackage[svgnames]{xcolor}
\usepackage[3d]{luadraw}
\begin{document}

\begin{luadraw}{name=stool_full}
local r = 20
local g = graph3d:new{window={-20,20,-30,30}, viewdir={45,75},size={12,12}}

local R = function(z)
    return 8 + (z/7)^2
end

local L = parametric3d( function(t) return M(0,R(t),t) end,-20,20,20,false,1)[1]
table.insert(L, M(0,0,20)); table.insert(L,1,M(0,0,-20))
S = rotline(L, {Origin, vecK}, 360, 0, {nbdots=80})
local mode = function(f)
    local G = isobar3d(f)
    return  -math.sqrt(math.abs(G.x)*math.abs(G.y))
end
local palette = {
  {0.658, 0.474, 0.000}, 
  {0.714, 0.533, 0.000},
  {0.769, 0.592, 0.000},
  {0.824, 0.651, 0.000},
  {0.878, 0.710, 0.000},
  {0.910, 0.769, 0.000},
  {0.937, 0.812, 0.000},
  {0.960, 0.855, 0.000},
  {0.980, 0.910, 0.000},
  {1.000, 0.950, 0.000} } --yellow
g:Dfacet(S, {
    usepalette = {palette,mode},
    mode=mShadedOnly,
    backcull = true
})
g:Show()
\end{luadraw}
\end{document}
