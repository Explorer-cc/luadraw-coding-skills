\documentclass{standalone}
\usepackage[3d]{luadraw}
\usepackage[svgnames]{xcolor}
\begin{document}

\begin{luadraw}{name=light_cone}
    local g = graph3d:new{
        window3d = {-5, 5, -5, 5, -5, 5},
        window = {-5,5,-1,5},
        viewdir = {0, 75},
        size = {10, 10}
    }
    local R, H = 3, 4
    -- calcul of the two sides, not so easy...
    local pi = math.pi 
    local f = function(t) return pt3d.det( g.Normal, Mc(R,t,H), Mc(R,t+pi/2,0) ) end
    local S = solve(f,-pi,pi)
    local t1, t2 = table.unpack(S)
    local A, B = Mc(R,t1,H), Mc(R,t2,H)
    
    g:Dcone(Origin, H*vecK, R, {mode=mWireframe, edgecolor="black",color="Pink"})

    local v_null = 0.6*Mc(R, math.rad(90), H)
    g:Dpolyline3d({Origin, v_null}, false, "blue, line width=1.2pt, -Stealth")
    g:Dpolyline3d({B, Origin, A}, "red")
    g:Ddots3d({A,B}, "red")
    g:Show()
\end{luadraw}

\end{document}
