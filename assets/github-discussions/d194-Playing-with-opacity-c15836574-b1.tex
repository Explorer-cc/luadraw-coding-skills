\documentclass[border = 3pt]{standalone}

\usepackage[svgnames]{xcolor}
\usepackage[3d]{luadraw}
\usetikzlibrary{shadings}

\begin{document}

\begin{luadraw}{name = vector-field-two-grids}
require 'luadraw_palettes'
require 'luadraw_shadedforms'
require 'luadraw_fields'

local mixpalette = function(pal, percent, color)
-- pal = palette of colors (rgb tables)
-- percent in [0;100]
-- color = rgb table
    color = color or White
    local mix = {}
    for _,c in ipairs(pal) do
        local str, tbl =  mixcolor(c,percent,color,100-percent) -- str = string format, tbl = table format
        table.insert(mix, tbl)
    end
    return mix
end

local f = function(x, y)
  return {1, x^2 + y^2 - 1}
end

local f_modulus = function(x, y)
  local A = f(x, y)
  return cpx.abs(Z(A[1],A[2]))
end

local xmin, xmax, ymin, ymax = -3, 3, -3, 3

local graphview = graph:new{
  window = {xmin - 0.5, xmax + 1.5, ymin - 0.5, ymax + 0.5},
  size   = {10, 10},
  bbox   = false
}

graphview:Dshadedrectangle(   
  xmin, xmax, ymin, ymax, 
  mixpalette(palJet, 50),
  {
    values = f_modulus,
    bar    = "right",
    baroptions
  }
)

graphview:Dvectorfield(
  f,
  {
    view         = {xmin, xmax, ymin, ymax},
    draw_options = "-stealth"
  }
)

graphview:Dgradbox(
  {
    Z(xmin, ymin),
    Z(xmax, ymax),
    1, 1
  },
  {
    title = "Vector field $f(x,y) = (1, x^2 + y^2 - 1)$"
  }
)

graphview:Show()
\end{luadraw}

\end{document}
