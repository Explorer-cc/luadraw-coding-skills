\documentclass{standalone} % compile with -enable-write18
\usepackage[svgnames]{xcolor}
\usepackage[3d]{luadraw}
\usepackage{fouriernc}
\usepackage{animate}

% global variables
\begin{luacode*}
nbimages = 36
name = "shadow"
basename = cachedir..name
\end{luacode*}

% images creation, use exec=false to avoid recreating images
\begin{luadraw}{auto=false,exec=true}
local z0 = -3
local g = graph3d:new{window3d={-5,8,-5,8,z0,5}, adjust2d=true,bbox=false,viewdir=perspective("yz",0.35,60), bg="Beige",size={10,10}}

-- declarations
require 'luadraw_polyhedrons'
local light = M(-5,-5,3.5)
local center = M(-1,-1,-1)
local proj = function(A)
    return proj3dO(A,{z0*vecK,vecK},A-light)
end
local P1, P = cube(center, M(-1,-1,z0))
local shadow, beam, beamV, beamH
local theta = linspace(0,360,nbimages+1) 

-- create the images
for k = 1, nbimages do 
    -- draw image number k here
    t = theta[k]; 
    P = rotate3d(P1,t,{ M(-1,-1,z0),vecK})
    shadow = cvx_hull3d( ftransform3d(P.vertices,proj) )[1] -- here we know that the shade is convex
    if not g:Isvisible(shadow) then shadow= reverse(shadow) end
    beam = pyramid( shadow,light,true)
    beamV,beamH = g:Classifyfacet(beam)
    
    g:Dfacet(g:Plane2facet({z0*vecK,vecK}), {color="lightgray", contrast=0})
    g:Dseg3d({M(-5,-5,z0), light})
    g:Dballdots3d(light,"Yellow",2)
    g:Dpolyline3d( shadow, true, "draw=none,fill=DarkGray,fill opacity=0.9") 
    g:Dfacet(beamH, {color="yellow",opacity=0.4,mode=mShadedOnly} )
    g:Dpoly(P, {color="cyan",mode=mShadedHidden})
    g:Dfacet(beamV, {color="yellow",opacity=0.4,mode=mShadedOnly} )

    -- save image number k
    g:Savetofile(basename..k..".tkz") -- do not modify
    g:Cleargraph() -- do not modify
end
\end{luadraw}
