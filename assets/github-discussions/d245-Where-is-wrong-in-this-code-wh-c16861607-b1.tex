\documentclass[border=2 mm]{standalone}
\usepackage[svgnames]{xcolor}
\usepackage[3d]{luadraw}
\usepackage{fouriernc}
\begin{document}
\begin{luadraw}{name=case5}
local ld = luadraw
local Z = ld.cpx.Z
local g = ld.graph:new{window={-2,4,-2,6}, size={10,10}}
g:Daxes({0,1,1}, {grid=true, arrows="-Stealth", legend={"$x$","$y$"}})
local f = function(x) return x^2-2*x end
g:Ddomain1(f,{ x={1,3},draw_options="draw=none,fill=pink,fill opacity=0.6"}) 
g:Dpolyline({{1,Z(1,f(1))}, {3,Z(3,f(3))}},"line width=0.6pt")
g:Dline({0,1}, "blue, line width=0.8pt") 
g:Dcartesian(f, {draw_options="red, line width=0.8pt"}); 
g:Show()
\end{luadraw}
\end{document}
