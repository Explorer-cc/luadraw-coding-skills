\documentclass[border=5pt]{standalone}
\usepackage[3d]{luadraw}
\usepackage[svgnames]{xcolor}
\begin{document}

\begin{luadraw}{name=sphere}
    local g = graph3d:new{
        window3d={-4,4,-4,4,-4,4},
        window={-5,5,-5,5},
        size={10,10}, bbox=false
    }
    local P = sphere(Origin,4)
    local S = {Origin, 4}
    g:Rotate3d(45,{Origin,M(1,1,1)})
    local plane = {M(0,0,-2.5), -vecK}
    local head = 4*vecK
    local a, b, c = table.unpack(map(toPoint3d,polyreg(0,1.7,3)))
    local a1,a2 = interDS( {head,head-a}, S)
    local b1,b2 = interDS( {head,head-b}, S)
    local c1,c2 = interDS( {head,head-c}, S)
    local I, r, n = interPS( plane, S)
    local P1 = cutpoly(P,plane)
    g:Dpoly(P1,{hiddenstyle="solid", hiddencolor="LightGray"})
    g:Dpolyline3d({{a1,a2},{b1,b2},{c1,c2}}, "line width=2.4pt")
    g:Dcylinder(head,vecK/2,0.25, {color="SlateGray",hiddenstyle="noline"})
    local L = g:Proj3d(border(P1))
    g:Beginclip( polyline2path(L), true)
        g:Dcircle3d(I+0.15*M(1,1/2,0),r,n,"fill=gray, even odd rule")
    g:Endclip()
    g:Show()
\end{luadraw}
\end{document}
