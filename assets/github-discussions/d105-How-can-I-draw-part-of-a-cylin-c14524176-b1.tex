\documentclass{standalone}
\usepackage[svgnames]{xcolor}
\usepackage[3d]{luadraw}
\begin{document}
\begin{luadraw}{name=half_cylinder}
local g = graph3d:new{window3d = {-1,4,-1,4,-1,2},size={10,10},viewdir={30,60}}
local C1 = cylinder(Origin,1,3*vecI,100,true)
local C2 = cylinder(Origin,1,3*vecJ,100,true)
C1 = cutpoly(C1,{Origin,vecJ},true)
C2 = cutpoly(C2,{Origin,vecI},true)
C1 = cutpoly(C1,{Origin,vecK},true)
C2 = cutpoly(C2,{Origin,vecK},true)
C1 = cutpoly(C1,{Origin,vecI-vecJ})
C2 = cutpoly(C2,{Origin,-vecI+vecJ})

g:Dscene3d(
    g:addWall({Origin,-vecI+vecJ}),
    g:addPoly(C1, {color="LightBlue"}),
    g:addPoly(C2, {color="LightBlue"}),
    g:addPolyline( concat(border(C1),border(C2)) ),
    g:addAxes(Origin,{arrows=1})
)
 g:Show()
\end{luadraw}
\end{document}
