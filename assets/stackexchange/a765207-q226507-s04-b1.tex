\documentclass[border=5pt]{standalone}% compile with lualatex only
\usepackage[svgnames]{xcolor}
\usepackage[3d]{luadraw}%https://github.com/pfradin/luadraw
\usepackage{fourier-otf}
%https://tex.stackexchange.com/questions/226507/pgf-draw-longitudinal-arcs-in-3d-axis-environment
\begin{document}
\begin{luadraw}{name=longitudinal_arcs}
local ld = luadraw
local pt3d = ld.pt3d
local Origin, vecI, vecJ, vecK = pt3d.Origin, pt3d.vecI, pt3d.vecJ, pt3d.vecK
local g = ld.graph3d:new{viewdir={25,60}, margin=0, bg="gray"}
ld.Hiddenlines=true; ld.Hiddenlinestyle = "dashed"
require 'luadraw_spherical'
g:Define_sphere({radius=4, color="SteelBlue", edgestyle="noline", arrowBstyle="-stealth"})
-- drawing
for k = 1, 6 do g:DSbigcircle({3*vecK, ld.sM(k*30,90)}, {color="white"}) end -- longitudinal
for k = 1, 9 do g:DScircle({ld.sM(0,k*18), vecK}, {color="white"}) end -- parallels
g:DSpolyline({{-5*vecI,5*vecI},{-5*vecJ,5*vecJ},{-5*vecK,5*vecK}}, {arrows=1, width=6}) --axes
g:Dspherical()
g:Dlabel3d("$\\sigma_x$",5*vecI,{pos="W"}, "$\\sigma_y$",5*vecJ,{pos="E"},"$\\sigma_+$",5*vecK,{pos="N"})
g:Show()
\end{luadraw}
\end{document}
