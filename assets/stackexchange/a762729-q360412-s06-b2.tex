\documentclass[border=5pt]{standalone}% compile with lualatex only
\usepackage[svgnames]{xcolor}
\usepackage[3d]{luadraw}%https://github.com/pfradin/luadraw
\usepackage{fourier-otf}
%%https://tex.stackexchange.com/questions/360412/drawing-cylinder-in-3d-plane
\begin{document}

\begin{luadraw}{name=two-cylinders2}
local ld = luadraw
local M = ld.pt3d.M
local g = ld.graph3d:new{  }
local R, z0 = 3, 3
ld.Hiddenlinestyle = "dashed"
-- cylinders
g:Dcylinder(M(0,1.5,-z0),1,M(0,1.5,z0),{edgestyle="dashed"}) -- small one
g:Dcircle3d(M(0,1.5,z0),1,M(0,0,1)) -- top of small cylinder
g:Dcylinder(M(0,0,-z0),R,M(0,0,z0)) -- large one
-- axes
g:Dpolyline3d({{M(3,0,-z0),M(5,0,-z0)}, {M(0,3,-z0), M(0,5,-z0)}, {M(0,0,z0),M(0,0,6)}},"-stealth")
g:Dpolyline3d({{M(0,0,-z0),M(5,0,-z0)}, {M(0,0,-z0), M(0,5,-z0)}, {M(0,0,-z0),M(0,0,6)}},"dashed")
g:Dcrossdots3d({M(0,0,z0),M(0,0,1)})
-- labels
g:Dlabel3d( "$U_n$",M(0,0,0),{pos="W"}, "$W$",M(1,2,0.5),{pos="center"}, 
   "$\\prime x$", M(5,0,-z0),{pos="S"}, "$\\prime y$", M(0,5,-z0),{},
    "$|z_n|$",M(0,0,5.25),{pos="E"}, "$U$", M(3,-0.5,0.5),{pos="center"},
    "$\\prime W$", M(2,2,-3),{pos="S",dist=0.25},"$\\prime U$",M(0,0,-z0),{pos="S"} )
g:Show()
\end{luadraw}
\end{document}
