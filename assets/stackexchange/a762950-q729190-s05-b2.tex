\documentclass[border=5pt]{standalone}% compile with lualatex only
\usepackage[svgnames]{xcolor}
\usepackage[3d]{luadraw}%https://github.com/pfradin/luadraw
\usepackage{fourier-otf}
%https://tex.stackexchange.com/questions/729190/create-a-paraboloid-and-a-right-circular-cone-opening-upward-with-vertex-at-the
\begin{document}

\begin{luadraw}{name= paraboloid-cone}
local ld = luadraw
local pt3d = ld.pt3d
local M, Mc = pt3d.M, pt3d.Mc
local pi = math.pi
local g = ld.graph3d:new{ window3d={0,3,0,4,0,5}, window={-3,12.5,-1,5.5}, 
    viewdir={-50,65}, size={16,8}, bg="lightgray"}
g:Labelsize("footnotesize")
local P = ld.surface( function(u,v) return Mc(u,v,u^2) end, 0,2,pi,-pi,{30,37})
local C = ld.surface( function(u,v) return Mc(u,v,u) end, 0,3,pi,-pi,{2,37})

local draw_surf = function(S, color)
    local Pv, Ph = g:Classifyfacet(S)
    g:Dpolyline3d({M(0,4,0),M(0,0,0), M(3,0,0)}, "stealth-stealth")
    g:Dpolyline3d( ld.border(Ph), 
        "left color="..color.."!20, right color="..color.."!50, middle color="..color.."!18")
    g:Dpolyline3d({M(0,0,0), M(0,0,5.5)}, "-stealth")
    g:Dpolyline3d( ld.border(Pv), 
        "left color="..color.."!50, right color="..color..", middle color="..color.."!10")
    g:Dpolyline3d({{M(0,0,0),M(0,0,5)},{M(0,4,0),M(0,0,0), M(3,0,0)}}, "dashed")
    g:Dlabel3d("$z$",M(0,0,5.5),{pos="N"},"$y$",M(0,4,0),{pos="E"},"$x$",M(3,0,0),{})
end

draw_surf(P, "SteelBlue"); g:Shift(8); draw_surf(C,"Crimson")
g:Show() 
\end{luadraw}
\end{document}
