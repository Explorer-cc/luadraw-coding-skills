\begin{document}
\begin{luadraw}{name=graph-of-a-projection-function}
local i = cpx.I
local g = graph3d:new{window3d={-1,4,-1,3,-1,3}, window={-3,3,-3,3},bbox=false, size={12,12}}
Hiddenlinestyle = "dashed"
local style = "draw=none,pattern color=gray, pattern = north west lines"
g:Labelsize("footnotesize"); g:Linewidth(6)
local P1 = parallelep(M(0,-1,0),vecI,2*vecJ,vecK)
local P2 = parallelep(M(2,-1,0),vecI,2*vecJ,2*vecK)
local u, v, w = g:Proj3dV(vecI), g:Proj3dV(vecJ), g:Proj3dV(vecK)
-- drawing axes
g:Dgradline({0,u}, {limits={-1,4}, arrows='->', tickdir=i, labelpos="top", labelstyle="S", legend="$x$", legendpos=1})
g:Dgradline({0,v}, {limits={-1,3}, originpos="none", arrows='->', tickdir=i, labelpos="top", labelstyle="S", legend="$y$", legendpos=1})
g:Dgradline({0,w}, {limits={0,3}, originpos="none", arrows='->', tickdir=v, labelpos="top", labelstyle="W", legend="$z$", legendpos=1})
g:Dpolyline3d( {{vecJ,vecJ+4*vecI},{-vecJ,-vecJ+4*vecI}, {M(2,1,2), M(0,1,2),M(0,0,2)}}, "dotted")
-- drawing parallelepiped
for _, P in ipairs({P1,P2}) do
    g:Dpolyline3d( getfacet(P,1), style)
    g:Dpoly(P,{mode=mWireframe})
    g:Dpolyline3d( getfacet(P,2), style)
end
g:Show()
\end{luadraw}
\end{document}
