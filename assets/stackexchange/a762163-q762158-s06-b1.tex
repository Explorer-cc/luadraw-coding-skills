\documentclass[tikz,border=3.14mm]{standalone}
\usepackage[svgnames]{xcolor}
\usepackage[3d]{luadraw}%https://github.com/pfradin/luadraw
\begin{document}
    \begin{luadraw}{name=tangent}
        local g = graph3d:new{viewdir={30,60}, window={-4,4,-3,7}, size={12,12}, bbox=false}
        local pi, sin, cos = math.pi, math.sin, math.cos
        local f = function(u, v) 
            return Mc(3+sin(v),u,u+cos(v)) -- cylindrical coordinates
        end
        local h = 1e-6
        local dfdu = function(u,v) return (f(u+h,v)-f(u-h,v))/h/2 end
        local dfdv = function(u,v) return (f(u,v+h)-f(u,v-h))/h/2 end
        local S = surface(f, 2*pi, 0, 0, 2*pi)
        -- Inverting the bounds of a parameter to correctly orient the surface (light color inside)
        g:Dfacet(S, {mode=mShadedOnly,color="blue"})
        local u, v = pi/2, 0
        local A, v1, v2 = f(u,v), pt3d.normalize(dfdu(u,v)), pt3d.normalize(dfdv(u,v))
        g:Ddots3d(A); g:Dpolyline3d( {{A,A+v1}, {A,A+v2}}, "-stealth,thick")
        g:Show()
    \end{luadraw}
\end{document}
