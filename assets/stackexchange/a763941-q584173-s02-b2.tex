\documentclass{standalone}
\usepackage[svgnames]{xcolor}
\usepackage[3d]{luadraw}
\usepackage{fourier-otf}
\begin{document}
\begin{luadraw}{name=quadratic_surface}
local ld = luadraw
local pt3d = ld.pt3d
local M, Origin, vecI, vecJ, vecK = pt3d.M, pt3d.Origin, pt3d.vecI, pt3d.vecJ, pt3d.vecK
local sin, cos, pi, sqrt = math.sin, math.cos, math.pi, math.sqrt
local g = ld.graph3d:new{
    window3d={-1.25,1.25,-0.25,2.25,0,6,1,1,0.5}, 
    adjust2d=true,size={10,10},
    viewdir={"central",60,50}
}

-- f(x,y) = 14*x^2 + 20*x*y - 22*x + 11*y^2 - 24*y + 14
-- f(x,y) = 14*(x+x0)^2 + 20*(x+x0)*(y+y0) + 11*(y+y0)^2 + 49/54, with x0=1/54 and y0=28/27

local x0, y0 = 1/54, 28/27
local T = M(x0,y0,0) -- translation vector

-- f(x,y) = 14*X^2 + 20*X*Y + 11*Y^2 + 49/54, with X=x+x0 and Y=y+y0 (translation)
-- f(x,y) = [X,Y]^T*M*[X,Y]+49/54 with the matrix M = [[14,10], [10,11]] 
-- M eigen values : (25+sqrt(409))/2 and (25-sqrt(409))/2
-- M eigen vectors [3+sqrt(409),10] and [3-sqrt(409),10]

local s = sqrt(409)
local a, b = (25-s)/2, (25+s)/2 -- eigen values
local U1, U2 = pt3d.normalize( M(3-s,20,0)), pt3d.normalize( M(3+s,20,0)) -- associated eigen vectors
-- f(x,y) = z becomes a*X^2+b*Y^2 = z in {T, U1, U2, vecK}, 
-- f(x,y) = z becomes a*X^2+b*Y^2 = Z in {T1, U1, U2, vecK} with T1 = T+49/54*vecK
local T1, T2 = M(x0,y0,49/54), M(x0,y0,6)
-- so we have a new parametrization in {T1, U1, U2, vecK} 

local inva, invb = 1/sqrt(a), 1/sqrt(b)
local p = function(r,t)
    return M(r*inva*cos(t), r*invb*sin(t), r^2)
end

local S = ld.surface(p,0, sqrt(6-49/54),-pi,pi,{30,50})
S = ld.mtransform3d(S, {T1,U1,U2,vecK}) -- change of coordinates from {T1, U1, U2, vecK} to {Origin, vecI, vecJ, vecK}.

g:Labelsize("footnotesize")
g:Dboxaxes3d({drawbox=true, grid=true, gridcolor="LightGray", xgradlimits={-1,1},
    ygradlimits={0,2}, xstep=0.5, ystep=0.5, fillcolor="WhiteSmoke", xlegend="$s$", ylegend="$t$"})
g:Dmixfacet(S,  {edgecolor="gray", edgewidth=1, usepalette={ld.palGasFlame,"z"}})
g:Dline3d( {T2,U1},"",true); g:Dline3d( {T2,U2},"",true); g:Dcrossdots3d({T,vecK},nil,0.5)
g:Dpolyline3d( {{T2,T2+U1}, {T2,T2+U2}}, "thick,-stealth") 
g:Dpolyline3d({T,T2},"dashed"); g:Dangle3d(T2+U1,T2,T2+U2,0.2,"thick")
g:Dlabel3d("$T$", T, {pos="E"}, "$U_1$", T2+U1,{pos='N'}, "$U_2$",T2+U2,{pos="W"})
g:Show()
\end{luadraw}
\end{document}
