\documentclass[10pt]{standalone}% compile with lualatex only
\usepackage[svgnames]{xcolor}
\usepackage{fouriernc}
\usepackage[3d]{luadraw}%https://github.com/pfradin/luadraw
%https://tex.stackexchange.com/questions/750541/pyramid-with-colors
\begin{document}
\begin{luadraw}{name=pyramid_with_colors}
local g = graph3d:new{ window3d={-2,5,-2,5,-6,3},window={-6,9,-9,3},viewdir={30,60}, size={10,10}}
g:Linejoin("round")
local A,B,C,D,V = M(-1,-2,2), M(-1, 4,2), M(-1, 4,-6), M(-1,-2,-6), M( 5, 4,-6)
local P = pyramid({A,B,C,D},V)
local F = proj3dO(getfacet(P),{-6*vecK,vecK},-vecK+vecJ-2*vecI) -- oblique projection onto the base plane
g:Dpolyline3d( g:Plane2facet({-6*vecK,vecK}),true) -- the base plane
g:Dboxaxes3d()
g:Dfacet(F,{mode=5,backcull=true,color="LightGray!30!white",opacity=0.5}) -- shadow
g:Dpoly(P,{color="cyan"}) -- pyramid
g:Show() 
\end{luadraw}
\end{document}
