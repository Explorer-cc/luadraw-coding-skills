\documentclass[border=5pt]{standalone}% compile with lualatex only
\usepackage[svgnames]{xcolor}
\usepackage[3d]{luadraw}%https://github.com/pfradin/luadraw
%https://ctan.org/pkg/luadraw
\usepackage{fourier-otf}
%https://tex.stackexchange.com/questions/765325/how-to-reproduce-the-correct-intersection-curve-of-two-3d-surfaces-using-pgfplot
\begin{document}
\begin{luadraw}{name=two_surfaces}
local ld = luadraw
local pt3d = ld.pt3d
local Origin, vecI, vecJ, vecK, M, Mc = pt3d.Origin, pt3d.vecI, pt3d.vecJ, pt3d.vecK, pt3d.M, pt3d.Mc
local ch, sh, pi = math.cosh, math.sinh, math.pi
local g = ld.graph3d:new{ window3d={-3,3,-3,3,-3,4}, viewdir={110,60}, 
    window={-4,4,-4.5,4.5}, size={12,12}, bg="lightgray"}
g:Linewidth(6)
local f1 = function(x,y) 
    return (x^2+x*y-y)/3
end
local f2 = function(x,y) 
    return (2*x^2+3*x*y-2*y)/3
end
local p3 = function(t)
    return M(t, t^2/(1-2*t), t^3/(3*(2*t-1)))
end
local L = ld.parametric3d(p3,-3,3,100,true)
L = ld.clippolyline3d(L, g:Box3d()) -- clip with the current 3D window
local S1, wall = ld.cartesian3d(f1, -3,3,-3,3,{25,25},"xy")
local S2 = ld.cartesian3d(f2, -3,3,-3,3)

local function visible_function(A)
-- Indicate whether point A on the green surface is on the visible side or not.
    local x, y = A.x, A.y
    local h = 1e-6
    local N = pt3d.prod( M(x+h,y,f2(x+h,y))-A, M(x,y+h,f2(x,y+h))-A )
    return (g:Cosine_incidence(N,A) > 0) 
end

local Lv, Lh = ld.split_points_by_visibility(L,visible_function)
g:Dscene3d(
    g:addWall(wall),
    g:addFacet(S1, {color="Crimson"}),
    g:addFacet(S2, {color="green"}),
    g:addAxes(Origin, {arrows=1, hidden=true})
    )
g:Dpolyline3d(Lv, "blue, line width=0.8")
g:Dpolyline3d(Lh, "blue, dashed, line width=0.6")
g:Show()
\end{luadraw}
\end{document}
