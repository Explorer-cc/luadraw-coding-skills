\documentclass{standalone}%compile with lualatex only
\usepackage[svgnames]{xcolor}
\usepackage{luadraw}%https://github.com/pfradin/luadraw/
\usetikzlibrary{shadings}  %use of a particular shading
%https://tex.stackexchange.com/questions/139114/differential-equation-direction-plot-with-pgfplots
\begin{document}
\begin{luadraw}{name=champ}
local x1,x2,y1,y2 = -3,3,-3,3
local g = graph:new{window={x1-0.5,x2+0.5,y1-0.5,y2+0.5},bg="",size={10,10}}

local f = function(x,y) -- éq. diff. y'= x^2+y^2-1=f(x,y)
    return x^2+y^2-1
end

local A = Z(0,1/2) -- initial condition
local n = 20
local deltaX, deltaY, long = (x2-x1)/n, (y2-y1)/n
local long = math.min(deltaX,deltaY)-0.1
local values, key = {}
local vectors, v = {}
local Min, Max = math.huge, -math.huge
for i = 0, n do
    local x = x1+deltaX*i
    local x1 = x+deltaX/2
    for j = 0, n do
        local y = y1+deltaX*j
        local y1 = y+deltaY/2
        key = i.."/"..j
        local img = f(x,y)
        if img < Min then Min = img end
        if img > Max then Max = img end
        values[key] = img
        if (i<n) and (j<n) then
            local v = Z(1,f(x1,y1))
            v = v/cpx.abs(v)*long -- normalization of v
            table.insert(vectors, {Z(x1,y1), Z(x1,y1)+v} )
        end
    end
end
local pal = {Purple,Indigo,Blue,Green,Yellow,Orange,Red}
local left_colors, right_colors, color = {}, {}
for j = 0,n do 
    key = "0/"..j
    color = mixcolor(palette(pal,(values[key]-Min)/(Max-Min),true),0.5,White,0.5) -- to replace the *fill opacity=0.5" option
    table.insert(right_colors, color)
end
-- paint
local eps =  1e-2 -- to widen the squares a little
for i = 0, n-1 do
    local x = x1+deltaX*i
    left_colors = right_colors
    right_colors = {}
    for j = 0,n do 
        key = (i+1).."/"..j
        color = mixcolor(palette(pal,(values[key]-Min)/(Max-Min),true),0.5,White,0.5)
        table.insert(right_colors, color)
    end
    for j = 0, n-1 do
        local y = y1+deltaX*j
        local color0 = left_colors[j+1]
        local color1 = right_colors[j+1]
        local color2 = right_colors[j+2]
        local color3 = left_colors[j+2]
        local options = "upper left="..color3..",upper right="..color2..",lower left="..color0..",lower right="..color1
        g:Dpolyline({Z(x-eps,y-eps),Z(x+deltaX+eps,y-eps),Z(x+deltaX+eps,y+deltaY+eps),Z(x-eps,y+deltaY+eps)}, "draw=none,fill opacity=0.5,"..options)
    end
end
g:Dgradbox({Z(x1,y1),Z(x2,y2),1,1} ,{originloc=0,originnum={0,0},nbsubdiv={3,3}})
g:Dpolyline(vectors, "-latex,darkgray")
local S = odesolve(f, A.re, A.im, x1,x2,150)
local L = {}
for k = 1, #S[1] do table.insert(L, Z(S[1][k],S[2][k])) end
L = clippolyline(L,-2.5,2.4,y1,y2)[1] -- L is the solution curve
-- preparations for drawing the curve
local wd = 8 -- width of the curve
local fillrec = function(z1,z2,c1,c2) -- shaded rectangle from z1 to z2
    local v = cpx.I*(z2-z1)
    v = v/cpx.abs(v)
    local ep = wd/20*pt*v
    local angle = cpx.arg(v)*rad
    if angle < 0 then angle = angle+180
    elseif angle > 180 then angle = angle-180 end
    g:Drectangle(z1-ep,z1+ep,z2+ep, "draw=none,left color="..c1..",right color="..c2..",shading angle="..strReal(angle))
end
local z2,z1 = L[1]
local m2,m1 = f(z2.re,z2.im)
local c2, c1 = palette(pal, (m2-Min)/(Max-Min))
-- now we draw the curve as succession of shaded rectangles
for k = 2, #L do
    z1 = z2; z2 = L[k]; m1 = m2; m2 = f(z2.re,z2.im)
    c1 = c2; c2 = palette(pal, (m2-Min)/(Max-Min))
    fillrec(z1,z2,c1,c2)
end
g:Dlabeldot("$A$", A, {pos="S"})
g:Show()
\end{luadraw}
\end{document}
