\documentclass[12pt]{standalone}
\usepackage[svgnames]{xcolor}
\usepackage[3d]{luadraw}
\usepackage{fouriernc}
\begin{document}
\begin{luadraw}{name=cutted_sphere1}
local g = graph3d:new{window={-5,5,-5,5},viewdir={30,60},size={10,10}}
g:Linejoin("round")
local R, h = 4, 3.5
local S = poly2facet( sphere(Origin,R,35,35) )
S = cutfacet(S, {h*vecK,-vecK}) -- to cut the top of S
S = cutfacet(S, {-h*vecK,vecK}) -- to cut the bottom of S
local rainbow = {Purple,Indigo,Blue,Green,Yellow,Orange,Red}
local facet_list = {}
for _,F in ipairs(S) do
    local z = isobar3d(F).z
    local index = (z+h)/(2*h)
    table.insert(facet_list,F)
    table.insert(facet_list, {color=palette(rainbow,index,true),opacity=0.85})
end
g:Dmixfacet( table.unpack(facet_list) )
g:Show()
\end{luadraw}

\begin{luadraw}{name=cutted_sphere2}
local g = graph3d:new{window={-5,5,-5,5},viewdir={30,60},size={10,10}}
g:Linejoin("round")
local R, h = 4, 3.5
local theta = math.asin(h/R)
local p = function(t) return M(0,R*math.cos(t),R*math.sin(t)) end --circle arc in yOz plane
local S = rotcurve(p, -theta, theta, {Origin,-vecK}, 0,360, {grid={35,35}})
local rainbow = {Purple,Indigo,Blue,Green,Yellow,Orange,Red}
local facet_list = {}
for _,F in ipairs(S) do
    local z = isobar3d(F).z
    local index = (z+h)/(2*h)
    table.insert(facet_list,F)
    table.insert(facet_list, {mode=5,color=palette(rainbow,index,true),contrast=0.5}) -- without edges
end
g:Dmixfacet( table.unpack(facet_list) )
g:Show()
\end{luadraw} 
\end{document}
