\begin{document}
\begin{luadraw}{name=q604334}
local ld = luadraw
local pt3d, cpx = ld.pt3d, ld.cpx
local M = pt3d.M
local g = ld.graph3d:new{ window3d={-8,8,-1,15,-8,8}, adjust2d=true, 
    viewdir={"central",55,60,50}, size={12,12}, bbox=false }
local scene = {}
local planes = {}

local function checkPlane(a,b,c,d, color) -- to add the plane ax+by+cz+d =0
    local P = ld.planeEq(a,b,c,d) -- P = {a point , a normal vector to the plane}
    table.insert(planes,P)
    table.insert(scene, g:addPlane(P, {color=color, twoside=false, scale=1, contrast=0, opacity=0.8 }) ) -- for drawing
end

checkPlane(2,1,-1,-11,"Gold") -- 2x+y-z-11=0
checkPlane(1,-3,0,20,"ForestGreen") -- x-3y+20= 0
checkPlane(4,2,5,-8,"SteelBlue") -- 4x+2y+5z-8 = 0
checkPlane(7,0,4,1,"Crimson") -- 7x+4z+1 = 0

local D = ld.interPP( planes[1], planes[2] )
if D ~= nil then
    local A = ld.interDP( D, planes[3] )
    if A ~= nil then
        local B, n = table.unpack( planes[4] )
        if math.abs( pt3d.dot(B-A,n) ) < 1e-6 then -- A is on all 4 planes.
            table.insert(scene, g:addDots(A, {color="red"})) 
        end
    end
end
g:Dboxaxes3d({grid=true, gridcolor="gray", fillcolor="lightgray", xyzstep=2, drawbox=true})
g:Dscene3d( table.unpack( scene ) )
g:Show()
\end{luadraw}
\end{document}
