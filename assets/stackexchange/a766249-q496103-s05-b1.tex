\documentclass{standalone}
\usepackage{luadraw}
\usepackage[svgnames]{xcolor}
\usepackage{fourier-otf}
%https://tex.stackexchange.com/questions/496103/how-can-i-draw-tangents-at-inflection-points-automatically
\begin{document}
    \begin{luadraw}{name=curve_tangents}
    local ld = luadraw
    local Z = ld.cpx.Z
    local g = ld.graph:new{window={-2,5,-2,5},size={7.5,7.5}}
    local f = function(x)
        -- return x^2*(3/2 +x*(-2/3 + x/12)) -- original function
        return 3*math.sin(x)^3+1.5 
    end
    local df2 = function(x)-- f''
        local h = 1e-3
        return (f(x+h)+f(x-h)-2*f(x))/h^2
    end
    g:Daxes({0,1,1},{
        arrows="very thick,-stealth", grid=true, xyticks={0,0},
        labelpos={"none","none"}, 
        legend={"$x$","$y$"},
        legendsep={0,0},
        legendpos={1,1}
    })
    local X = ld.solve(df2,-2,5)
    if X ~= nil then
        for _, x in ipairs(X) do
            g:DtangentC(f,x,"thick,purple"); g:Ddots(Z(x,f(x)),'purple')
        end
    end
    g:Dcartesian(f, {draw_options="thick,blue"})
    g:Show()
\end{luadraw}
\end{document}
