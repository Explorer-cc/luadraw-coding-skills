\documentclass[border=5pt]{standalone}% compile with lualatex only
\usepackage[svgnames]{xcolor}
\usepackage[3d]{luadraw}%https://github.com/pfradin/luadraw
\usepackage{fourier-otf}
%https://tex.stackexchange.com/questions/765311/how-can-i-draw-the-plane-and-intersection-curve-on-a-one-sheet-hyperboloid-in-ti
\begin{document}
\begin{luadraw}{name=hyperboloid_plane}
local ld = luadraw
local pt3d = ld.pt3d
local Origin, vecI, vecJ, vecK, M = pt3d.Origin, pt3d.vecI, pt3d.vecJ, pt3d.vecK, pt3d.M
local pi = math.pi
local g = ld.graph3d:new{ window3d={-3,3,-3,3,-2,2}, viewdir={45,60}, 
    window={-4,4,-3.5,3.5}, size={12,12}}
ld.Hiddenlines=true; ld.Hiddenlinestyle = "dashed"; g:Linewidth(6)
local P = {M(0,0,1), M(2,0,1)} -- 2X+z=1
local r = function(u,v) return math.cosh(u) end
local z = function(u,v) return math.sinh(u) end
local S, wall = ld.cylindrical_surface(r,z,-2,2,-pi,pi,{25,51},"zv") -- hyperboloid
local I = g:Intersection3d(S, P)
g:Dscene3d(
    g:addWall(wall), --partition walls
    g:addPlane(P, {color="pink", edge=true, scale=1, opacity=1}),
    g:addFacet(S, {color="Crimson", clip=true}),
    g:addPolyline(ld.concat(I.visible,I.hidden), {color="blue", width=8, clip=true}),
    g:addAxes(Origin, {arrows=1})
    )
g:Show()
\end{luadraw}
\end{document}
