\documentclass[border=5pt]{standalone}% compile with lualatex only
\usepackage[svgnames]{xcolor}
\usepackage[3d]{luadraw}%https://github.com/pfradin/luadraw
\usepackage{fourier-otf}
%https://tex.stackexchange.com/questions/503427/tikz-3dplot-angle-placed-cones-in-a-sphere
\begin{document}
\begin{luadraw}{name=cones_in_sphere}
local ld = luadraw
local pt3d = ld.pt3d
local M, O, I, J, K= pt3d.M, pt3d.Origin,  pt3d.vecI, pt3d.vecJ, pt3d.vecK
local g = ld.graph3d:new{ window={-2.25,2.25,-2.25,2.25}, viewdir={40,55},size={10,10}}
local poly = require 'luadraw_polyhedrons'
local P = poly.octahedron(O,3*K)
local colors = {"blue", "red"}
local S = ld.getfacet(P)
for k,F in ipairs(S) do
    table.insert(F, colors[1+k%2])
end
S = g:Sortfacet(S)
local I, r = ld.incircle3d( table.unpack(S[1]) )
for _,F in ipairs(S) do
    local color = table.remove(F)
    I = pt3d.isobar3d(F)
    g:Dcone(I,r,O,{color=color,opacity=0.7})
end
g:Dsphere(O,math.sqrt(3+r^2),{color="orange",mode=ld.mBorder, opacity=0.2})
g:Show()
\end{luadraw}
\end{document}
