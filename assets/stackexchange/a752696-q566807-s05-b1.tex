\documentclass[border=5pt]{standalone}% compile with lualatex only
\usepackage[svgnames]{xcolor}
\usepackage[3d]{luadraw}%https://github.com/pfradin/luadraw
%https://tex.stackexchange.com/questions/566807/how-to-plot-3d-convex-hull-using-tikz-pgf
\begin{document}
\begin{luadraw}{name=3d_convex_hull}
local g = graph3d:new{window3d={0,3,0,3,0,1},adjust2d=true,bbox=false,size={10,10}, viewdir={-30,50}}
local data = {
M(0,0,0.8),
M(1,0,0.56),
M(2,0,0.5),
M(3,0,0.75),
M(0,1,0.6),
M(1,1,0.3),
M(2,1,0.21),
M(3,1,0.3),
M(0,2,0.68),
M(1,2,0.22),
M(2,2,0.25),
M(3,2,0.4),
M(0,3,0.7),
M(1,3,0.5),
M(2,3,0.58),
M(3,3,0.9)}
local F = cvx_hull3d(data) -- F is a list of facets
g:Dboxaxes3d({grid=true,gridcolor="gray",fillcolor="LightGray",drawbox=false})
g:Dscene3d(  g:addFacet(F, {usepalette={palRainbow,"z"},edge=true,opacity=0.8} ) )
g:Show()
\end{luadraw}
\end{document}
