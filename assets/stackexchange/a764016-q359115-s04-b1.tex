\documentclass[border=5pt]{standalone}% compile with lualatex only
\usepackage[svgnames]{xcolor}
\usepackage[3d]{luadraw}%https://github.com/pfradin/luadraw
\usepackage{fourier-otf}
%https://tex.stackexchange.com/questions/359115/creating-a-surface-plot-of-a-function-on-the-simplex-using-pstricks-or-tikz
\begin{document}
\begin{luadraw}{name=surface_on_simplex, exec=true, auto=false}
local ld = luadraw
local pt3d = ld.pt3d
local M = pt3d.M
local O, I, J, K = pt3d.Origin, pt3d.vecI, pt3d.vecJ, pt3d.vecK
local sin, cos, deg= math.sin, math.cos, ld.deg

local g = ld.graph3d:new{ window3d ={0,1,0,1,0,1}, viewdir={-25,70}, window={-0.75,1.5,-0.5,1.5},
size={12,12} }

local function f(x,y,z)
    if x > 1/2 then return x/(2-x) 
    elseif z > 1/2 then return (1-z)/(2-z) 
    elseif y > 1/2 then return (1-y)/(2-y)
    else return 1/3
    end
end    

local A, B, C = M(1,0,0), M(0,1,0), M(0,0,1)
local G = (A+B+C)/3
local N = pt3d.normalize( pt3d.prod(B-A,C-A) ) --- normal vector
local mat = ld.matrix3dof( function(P) return ld.rotateaxe3d(P,N,K,G) end ) --transforms N to K

local function p(u,v)
    local D = u*A + (1-u)*( v*B + (1-v)*C ) -- D is in the triangle (A,B,C)
        return D + f(D.x,D.y,D.z)*N
end
local shading = function(f) -- f is a facet
    local I = pt3d.isobar3d(f)
    return pt3d.abs(I)
end

local S = ld.surface(p,0,1,0,1,{40,20})
g:Rotate3d(-120,{G,K}); g:Composematrix3d( mat ) -- to position the triangle
-- drawing
g:Dfacet({A,B,C}, {color="gray!20!white", contrast=0})
g:Dpolyline3d( {{O,1.25*I},{O,1.5*J},{O,1.25*K}}, "thick,-stealth")
g:Dfacet(S, {usepalette={ld.palRainbow,shading}, opacity=0.8, edgecolor="gray", edgewidth=1})
g:Dpolyline3d({{A,A+f(A.x,A.y,A.z)*N}, {B,B+f(B.x,B.y,B.z)*N}, {C,C+f(C.x,C.y,C.z)*N}})
g:Dpolyline3d({{(A+C)/2+N/3,A+N/3,(A+B)/2+N/3,(A+B)/2},{(A+C)/2, (A+C)/2+N/3}, {(B+C)/2,(B+C)/2+N/3}}, 'dashed')
g:Dpolyline3d({(A+B)/2+N/3,(A+C)/2+N/3,(B+C)/2+N/3},true, 'dashed')
g:Dpolyline3d({(A+B)/2,(A+C)/2,(B+C)/2},true, 'dashed')
g:Dlabel3d("$x$",1.25*I,{pos="N"}, "$y$",1.5*J,{pos="NE"},"$z$",1.25*K,{pos='N'},
    "$(0,0,1)$",C,{pos="E"}, "$(1,0,0)$",A,{pos="W"}, 
    "$(0,1,0)$",B,{pos="W",dist=0.1}, "$O$",O,{pos="S", dist=0}
)
g:IDmatrix3d()
g:Show()
\end{luadraw}
\end{document}
