\documentclass[border=5pt]{standalone}% compile with lualatex only
\usepackage[svgnames]{xcolor}
\usepackage[3d]{luadraw}%https://github.com/pfradin/luadraw
%https://ctan.org/pkg/luadraw
\usepackage{fourier-otf}
%https://tex.stackexchange.com/questions/487872/drawing-a-top-in-tikz
\begin{document}
\begin{luadraw}{name=drawing_top}
local ld = luadraw
local cpx = ld.cpx
local pt3d = ld.pt3d
local M, Origin, vecI, vecJ, vecK, Ms, Mc, deg = pt3d.M, pt3d.Origin, pt3d.vecI, pt3d.vecJ, pt3d.vecK, pt3d.Ms, pt3d.Mc, ld.deg

local g = ld.graph3d:new{ window={-3,4.5,-1.5,7}, size={12,12}, viewdir={30,70} }
require 'luadraw_decorations'
g:Labelsize("footnotesize")
local A, B, v1, v2 = M(0,1.5,4.5), Origin, M(0,1.5,-1.5), M(0,-0.5,-2)
local L = ld.path3d( {A,A+v1,B-v2,B,"b"})[1]
local S = ld.rotline(L, {Origin, vecK}, 0,360, {nbdots=37})
local N = Ms(1,90*deg,15*deg)/2
S = ld.rotateaxe3d(S, vecK, N)
local C, U, V = table.unpack( ld.rotateaxe3d({ld.pz(A), Mc(1,-80*deg,A.z), Mc(1,90*deg,A.z)}, vecK,N) )
local Sv, Sh = g:Classifyfacet(S)
local I, J, G = ld.interDP({Origin, vecK}, {C,N}), C+3*N, C-2.5*N
local Jz, K = ld.pz(J), ld.rotate3d(J, 25, {Origin, vecK} )
local r = pt3d.abs(J-Jz)
local T = pt3d.normalize( pt3d.prod(J-Jz,-vecK) )
-- drawing
g:Dpolyline3d( ld.border(Sv), "ball color=LightBlue")
g:Dcircle3d(C,A.y,N, "fill=LightBlue!80!black")
g:Dcylinder(C,A.y/4,C+N, {color="LightBlue!80!black", hiddenstyle="noline", gradside={100,50,100}, gradsection={80,90,100}})
g:Dseg3d( {G, C+N}, "orange!40!LightBlue, thick")
g:Darc3d(U,C,V,A.y/2,1, {normal=N, arc_options="DarkGreen,thick,-latex", label="$\\omega$"})
g:Dpath3d({J, Jz, vecK, "c"}, "dashed")
g:Dseg3d( {Origin, G}, "black!40!LightBlue,thick,-latex")
g:Dseg3d( {G, G-1.5*vecK}, "red!40!LightBlue,thick,-latex")
g:Dseg3d( {G,G+0.75*T}, "orange!40!LightBlue, thick,-latex")
g:Dpolyline3d({{Jz,J}, {C+N, J}}, "orange, thick, -latex")
g:Darc3d(J,Jz,K,r,1,"orange, thick, -latex")
g:Dpolyline3d({{Origin, 5*vecI},{Origin,5*vecJ}, {I, 7*vecK}}, "thick,-latex")
g:Dseg3d({Origin,I},"dashed")
g:Dseg3d({Jz,K},"dashed")
g:Darc3d(J,Jz,K,0.75,1)
g:Darc3d(vecK,Origin,J,1,1, {label="$\\theta$"})
g:Darc3d( Mc(1, -100*deg,J.z), Jz, Mc(1,-10*deg,J.z), r+0.25,1, {arc_options="-latex", label="$\\omega_p$"})
g:Ddots3d(G,"gray")
g:Dlabel3d( "$x$",5*vecI,{pos="W"}, "$y$", 5*vecJ,{pos="E"}, "$z$",7*vecK,{pos="N"},"$O$",Origin,{pos="S"},
    "$\\vec{L}$",J,{pos="SE"},"CM",G,{pos="NW"},
    "$\\vec{\\tau}=\\vec r\\times M\\vec g$",G+0.25*T,{pos="E", dist=0.15},
    "$M\\vec g$",G-1.5*vecK,{pos="E",dist=0}, "$\\vec r$",0.75*G,{pos="W"},
    "$L\\sin\\theta$",(J+2*Jz)/3,{pos="S"},
    "d$L=\\vec{\\tau}$d$t$", (J+K)/2, {pos="E"},"d$\\phi$",(Jz+K)/2,{pos="N"} )
g:Show()
\end{luadraw}
\end{document}
