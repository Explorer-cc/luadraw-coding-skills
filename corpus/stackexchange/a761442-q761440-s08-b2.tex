\begin{luacode*}
function background()
    local g = graph:new{
        margins={0,0,0,0}
        , size={18,18} 
        }
    g:Linecolor("lightgray")
    for j= 1, 5 do
        for k=1,10*j do
            g:Dcircle(Zp(j,k*math.pi/(5*j)),1)
        end
    end
    g:Sendtotex()
end
\end{luacode*}
\def\background{\directlua{background()}}%

\AddToHook{shipout/background}
{%
    \put(\paperwidth/2-9cm,-\paperheight/2-9cm){\background}%
}
