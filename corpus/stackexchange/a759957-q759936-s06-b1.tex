\documentclass[border=5pt]{standalone}% compile with lualatex only
\usepackage[svgnames]{xcolor}
\usepackage[3d]{luadraw}%https://github.com/pfradin/luadraw
\usepackage{fourier-otf}
%https://tex.stackexchange.com/questions/759936/how-to-draw-a-half-sphere-with-5-degree-lines-along-the-x-axis-in-tikz-pgf
\begin{document}

\begin{luadraw}{name=half-sphere2}
local pi, cos, sin = math.pi, math.cos, math.sin
local g = graph3d:new{size={10,10}, viewdir={25,60},margin={0,0,0,0}}
require 'luadraw_spherical'
Hiddenlinestyle = "dashed"
local R = 4
g:Define_sphere({radius=4, show=false})
local S = surface( function(u,v) return R*M(cos(v),sin(v)*cos(u),sin(v)*sin(u)) end, 0,-2*pi,0,pi/2,{37,10})
S = g:Classifyfacet(S)
g:Dpolyline3d({{-5*vecI,5*vecI}, {-5*vecJ,5*vecJ}, {-5*vecK,5*vecK}}, "line width=0.8pt,-stealth,Maroon")
g:Dpolyline3d( border(S), true, "fill=white,draw=none")
S = facet2poly(S)
local A, N = R*vecJ, vecK
for k = 1, 18 do
    g:DSarc({-A,A},1, {normal=N, width=2, color="gray",hidden=true})
    A,N = table.unpack( rotate3d( {A,N}, 10, {Origin,vecI}))
end
A = R*vecK
for k = 1, 9 do
    g:DScircle( {A,vecI}, {width=2, color="gray",hidden=true})
    A = rotate3d(A, 10, {Origin,vecJ})
end
g:DSarc({vecJ,-vecJ},-1,{normal=vecK, width=12, hidden=true})
g:DSarc({vecK,-vecK},1,{normal=vecJ, width=12, hidden=true})
g:DSstars( S.vertices, {fill="white",scale=0.5})
g:Dspherical()
g:Dpolyline3d({R*vecI,5.5*vecI}, "line width=0.8pt,-stealth,Maroon")
g:Dpolyline3d({{-5*vecI,5*vecI}, {-5*vecJ,5*vecJ}, {-5*vecK,5*vecK}}, Hiddenlinestyle..", Maroon")
g:Dlabel3d("$x$",5*vecJ,{pos="E",node_options="Maroon"}, "$z$", 5.5*vecI,{pos="W"}, "$y$",5*vecK,{pos="N"})
g:Show()
\end{luadraw}
\end{document}
