\documentclass[12pt]{standalone}% compile with lualatex only
\usepackage[svgnames]{xcolor}
\usepackage{fouriernc}
\usepackage[]{luadraw}%https://github.com/pfradin/luadraw
%https://tex.stackexchange.com/questions/739244/plotting-a-discontinuous-function-with-pgfplots-in-latex
\begin{document}
\begin{luadraw}{name=discontinous_function}
local g = graph:new{window={-0.1,5,-3,5}, size={12,12,0}}
g:Linejoin("round"); g:Linewidth(6)
local i = cpx.I
local h = function(t) return 1/math.log(t) end
local f = function(x) if (x<1/2) or (x>1) then return int(h,x,2*x) end end
g:Dline({0.5,cpx.I},"blue"); g:Dline({1,cpx.I},"blue")
g:Daxes({0,1,1},{arrows="-stealth"})
g:Dcartesian(f,{discont=true, nbdiv=10, draw_options="red"})
g:Show()
\end{luadraw}
\end{document}
