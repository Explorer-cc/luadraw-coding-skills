\documentclass[border=5pt]{standalone}% compile with lualatex only
\usepackage[svgnames]{xcolor}
\usepackage[3d]{luadraw}%https://github.com/pfradin/luadraw
\usepackage{fourier-otf}
%https://tex.stackexchange.com/questions/53276/draw-rounded-cube?rq=1
\begin{document}
\begin{luadraw}{name=rounded-cube}
local ld = luadraw
local pt3d = ld.pt3d
local M, vecI, vecJ, vecK = pt3d.M, pt3d.vecI, pt3d.vecJ, pt3d.vecK
local sqrt, sin, cos = math.sqrt, math.sin, math.cos

local g = ld.graph3d:new{ window={-4,4.5,-3,5},viewdir={35,70},bg="Bisque" }
ld.Hiddenlinestyle = "dashed"
local L, W, H = 4, 4, 4 --5.0, 2.4, 3.0 -- length, width, height
local c = 1 -- contrast
local bc = true -- backculling
local construction = true -- Construction lines or not
local color = "SlateGray"
local t = 0.8
local nb1, nb2 = 18, 12
local scene = {}

local function add_quarter_cylinder(center, r1, r2, axis, n)
  if pt3d.det(r1,r2,axis) < 0 then r1, r2 = r2, r1 end -- orientation
  local arc1, arc2 = {}, {}
  for k = 0,n do
    local u = k*math.pi/2/n
    local p = center + cos(u)*r1 + sin(u)*r2
    table.insert(arc1, p)
    table.insert(arc2, p+axis)
  end
  local poly = ld.prism(arc1,axis,false)
  table.insert(scene, g:addPoly(poly, {color=color, contrast=c, backcull=bc}))
  if construction then
    table.insert(scene, g:addPolyline( {arc1, arc2}, {hidden=true}))
  end
end

local function add_octant_sphere(center, r1, r2, r3, n)
    if pt3d.det(r1,r2,r3) < 0 then r1, r2 = r2, r1 end -- orientation
    local arc = {}
    for k = 0, n do
        local u = k*math.pi/2/n
        table.insert(arc, center + cos(u)*r3 + sin(u)*r1)
    end
    local poly = ld.rotline(arc, {center,r3}, 0, 90, {nbdots=n})
    table.insert(scene, g:addFacet(poly, {color=color, contrast=c, backcull=bc}))
    if construction then
        table.insert(scene, g:addPolyline({{center,center+r1}, {center,center+r2}, 
            {center,center+r3}}, {hidden=true}))
    end
end

local box_inside = ld.parallelep(M(0,0,0), L*vecI, W*vecJ, H*vecK)
local facets = {}
for _, F in ipairs( ld.getfacet(box_inside) ) do -- shift the box facets
    local G, u = pt3d.isobar3d(F)
    if G.z == H then u = t*vecK
    elseif G.z == 0 then u = -t*vecK
    elseif G.x == 0 then u = -t*vecI
    elseif G.x == L then u = t*vecI
    elseif G.y == 0 then u = -t*vecJ
    elseif G.y == W then u = t*vecJ
    end
    table.insert(facets, ld.shift3d(F,u) )
end

--to avoid unnecessary facet intersection calculations
table.insert(scene, g:addWall( ld.facet2plane(ld.getfacet(box_inside)) ))

table.insert(scene, g:addFacet(facets, {color=color,contrast=c, 
    backcull=bc, edge=construction, hidden=true}))
table.insert(scene, -- box inside
    g:addPoly( box_inside,{ color=color,contrast=c, backcull=bc, edge=construction, hidden=true }))
-- L
add_quarter_cylinder(M(0,W,H), t*vecJ, t*vecK, L*vecI, nb1)
add_quarter_cylinder(M(0,W,0), t*vecJ, -t*vecK, L*vecI, nb1)
add_quarter_cylinder(M(0,0,H), -t*vecJ, t*vecK, L*vecI, nb1)
add_quarter_cylinder(M(0,0,0), -t*vecJ, -t*vecK, L*vecI, nb1)
-- W
add_quarter_cylinder(M(L,0,H), t*vecI, t*vecK, W*vecJ, nb1)
add_quarter_cylinder(M(L,0,0), t*vecI, -t*vecK, W*vecJ, nb1)
add_quarter_cylinder(M(0,0,H), -t*vecI, t*vecK, W*vecJ, nb1)
add_quarter_cylinder(M(0,0,0), -t*vecI, -t*vecK, W*vecJ, nb1)
-- H
add_quarter_cylinder(M(L,W,0), t*vecI, t*vecJ, H*vecK, nb1)
add_quarter_cylinder(M(L,0,0), t*vecI, -t*vecJ, H*vecK, nb1)
add_quarter_cylinder(M(0,W,0), -t*vecI, t*vecJ, H*vecK, nb1)
add_quarter_cylinder(M(0,0,0), -t*vecI, -t*vecJ, H*vecK, nb1)

add_octant_sphere(M(L,W,H), t*vecI, t*vecJ, t*vecK, nb2)
add_octant_sphere(M(L,W,0), t*vecI, t*vecJ, -t*vecK, nb2)
add_octant_sphere(M(L,0,H), t*vecI, -t*vecJ, t*vecK, nb2)
add_octant_sphere(M(L,0,0), t*vecI, -t*vecJ, -t*vecK, nb2)
add_octant_sphere(M(0,W,H), -t*vecI, t*vecJ, t*vecK, nb2)
add_octant_sphere(M(0,W,0), -t*vecI, t*vecJ, -t*vecK, nb2)
add_octant_sphere(M(0,0,H), -t*vecI, -t*vecJ, t*vecK, nb2)
add_octant_sphere(M(0,0,0), -t*vecI, -t*vecJ, -t*vecK, nb2)

g:Dscene3d(table.unpack(scene))
g:Show()
\end{luadraw}
\end{document}
