\documentclass[border=2pt]{standalone}% compile with lualatex only
\usepackage[svgnames]{xcolor}
\usepackage[3d]{luadraw}%https://github.com/pfradin/luadraw
\usepackage{fourier-otf}
%https://tex.stackexchange.com/questions/501274/tikz-3dplot-how-to-draw-small-circles-of-a-sphere-in-a-given-angle?rq=1
\begin{document}
\begin{luadraw}{name=projstereo}
local ld = luadraw
local pt3d = ld.pt3d
local M = pt3d.M
require 'luadraw_spherical'
local sM = ld.sM
local g = ld.graph3d:new{ window={-9.5,14.5,-9,4}, size={16,8}, viewdir={-20,70}}
local O, R = M(0,0,0), 3
g:Define_sphere( {color="orange", opacity=0.7} )
local A, B, C, N, S = sM(-10,90), sM(0, 50), sM(50,65), sM(0,0), sM(0,180)
local P = ld.plane(A,B,C)
local P2 = {M(2,2,-R), M(0,0,1)}
local I, r = ld.interPS(P, {O,R} )
local P3 = {M(2,2,I.z), M(0,0,1)}
local F = {A,B,C}
local p = ld.projstereo(F, {O,R}, N, -R)
local p2 = ld.projstereo(F, {O,R}, N, I.z)
local J, r2 = ld.circumcircle3d( table.unpack(p) )
local K, r3 = ld.circumcircle3d( table.unpack(p2) )
local J1 = J + r2*pt3d.normalize(S-J)
local I1 = ld.inv_projstereo(J1, {O,R}, N)
local K1 = ld.interDP({N,J1-N}, P3) 
local T = ld.projstereo_Scircle(P, N, -R)
local T3 = ld.projstereo_Scircle(P, N, I.z)
-- drawing
g:Dplane(P2, M(0,1,0), 16, 26, "fill=lightgray")
g:DSpolyline( {{N,J}, {N,J1}}, {color="blue", width=8})
g:DScircle(P, {color="yellow", width=8})
g:DSstars(N, {fill="black"})
g:DSplane(P3, {scale=1.75, color="gray"})
g:DSpolyline( ld.path3d(T3), {color="blue", width=8})
g:DSpolyline( {{K1,K,I1}, {J1,J}}, {color="yellow", width=8})
g:DSdots( {K1,J,J1,I1,K} )
g:Dspherical()
g:Dpath3d(T, "line width=0.8")
g:Labelsize("footnotesize")
g:Dlabel3d("$N$",N,{pos="N"}, "$J$",J,{pos="NE"}, "$J_1$",J1,{}, "$S$",K,{pos="E"})
g:Show()
\end{luadraw}
\end{document}
