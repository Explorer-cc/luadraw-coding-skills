\begin{luadraw}{name=tikz-with-equation-regions}
local g = graph3d:new{window3d = {-1,1,-1,1,-1,1},window={-2,2,-2,2},bbox=false,viewdir={-120,50},size={10,10}}
g:Linewidth(1)
local cos, sin, pi = math.cos, math.sin, math.pi
local  p = function(u,v) return M(cos(u),sin(u+v),sin(v)) end
local s = surface(p,0,pi,pi/2,5*pi/2,{25,50})

g:Dboxaxes3d({grid=true,gridcolor="gray",fillcolor="lightgray",drawbox=true})
g:Dfacet(s, {color="Tomato",backcull=true})
g:Show()
\end{luadraw}
