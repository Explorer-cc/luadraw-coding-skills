\documentclass[border=5pt]{standalone}% compile with lualatex only 
\usepackage[svgnames]{xcolor} 
\usepackage[3d]{luadraw}%https://github.com/pfradin/luadraw
\begin{document}
\begin{luadraw}{name=line2tube}
  local g = graph3d:new{window={-6,6,-5,7},margin={0,0,0,0},bbox=false, size={10,10}, viewdir={30,60}}
  local L1 = {M(0,0,0), M(0,0,3),M(0,3,3),M(0,3,0)} -- polygonal line, do not close it here
  local tube1 = line2tube(L1, 0.75, {nbfacet=100, close=true}) -- option close=true to close the polygonal line
  g:Dmixfacet(tube1,{color="SteelBlue",mode=mShadedOnly, opacity=0.8})
  g:Show()
\end{luadraw}
\end{document}
