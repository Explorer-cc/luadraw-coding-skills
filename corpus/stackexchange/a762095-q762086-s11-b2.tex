\documentclass[border=5pt]{standalone}% compile with lualatex only
\usepackage[svgnames]{xcolor}
\usepackage[3d]{luadraw}%https://github.com/pfradin/luadraw
\usepackage{fourier-otf}
%https://tex.stackexchange.com/questions/762086/plot-a-domain-in-3d-with-tikz-for-a-triple-integral
\begin{document}
\begin{luadraw}{name= integral_domain}
local g = graph3d:new{window3d={-1,1,-1,1,-1,1}, adjust2d=true, 
    viewdir=perspective("central",30,60), size={10,10}, bg="white"}
local L = {Origin, M(0,1,-1), M(0,1,1)} -- the triangle
local z = -0.25 -- to choose a plane
local vz = math.abs(z)
local axis, plane = {Origin, vecK}, {z*vecK, vecK}
local S = rotline(L, axis, 90, 0, {close=true, nbdots=10})
table.insert(S, L)
table.insert(S, reverse( rotate3d(L, 90, axis) ))
local S1, S2 = cutfacet(S, plane)
local section 
if vz>1e-4 then
    section = {M(0,1,z),"m",z*vecK,M(-1,0,z),1,1,"ca",M(-vz,0,z),"l",z*vecK,M(0,vz,z),math.abs(z),1,"ca",M(0,1,z),"l","cl"}
else
    section = {M(0,1,z),"m",z*vecK,M(-1,0,z),1,1,"ca",M(-vz,0,z),M(0,1,z),"l","cl"}
end
g:Dboxaxes3d({grid=true, gridcolor="gray", fillcolor="lightgray"})
g:Dfacet(S2, {color="blue!40!white", opacity=0.6, edgewidth=1, edgecolor="blue"})
g:Dpath3d(section, "draw=none,pattern=north west lines")
g:Dfacet(S1, {color="blue!40!white", opacity=0.6, edgewidth=1, edgecolor="blue"})
g:Lineoptions("dotted",nil,6)
g:Dcylinder(-vecK,1,vecK, {mode=mWireframe})
g:Dpolyline3d({{vecK,-vecK}, {M(-1,0,1),M(1,0,1)}, {M(0,-1,1),M(0,1,1)}})
g:Show() 
\end{luadraw}
\end{document}
