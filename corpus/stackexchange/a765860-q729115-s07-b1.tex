\documentclass{standalone} %lualatex only
\usepackage[3d]{luadraw}
\usepackage[svgnames]{xcolor}
%https://tex.stackexchange.com/questions/729115/draw-a-m%c3%b6bius-trip-with-tangent-vectors
\begin{document}
\begin{luadraw}{name=mobius_normal_vectors}
local ld = luadraw
local pt3d = ld.pt3d
local M, Ms, Mc = pt3d.M, pt3d.Ms, pt3d.Mc
local sin, cos, pi = math.sin, math.cos, math.pi
local g = ld.graph3d:new{ window={-1.25,1.25,-0.75,0.75}, viewdir={-50,65}, 
    size={12,12}, bg="lightgray" }
local R, h = 1, 0.25
local r = function(u,t)
    return R + h*u*cos(t/2)
end
local z = function(u,t)
    return h*u*sin(t/2)
end
local s, wall = ld.cylindrical_surface(r,z, -1, 1, -pi, pi, {5,101}, "v")
local n, len = 30, 0.2
local L1, L2 = {}, {}
for k = 0, n-1 do
    local t1, t2 = 2*pi*k/n, pi*(2*k+1)/n
    local P1, P2 = Ms(R,t1,pi/2), Ms(R,t2,pi/2)
    local nv1 = pt3d.normalize( Mc(-0.5*sin(t1/2),t1,0.5*cos(t1/2)) )
    local nv2 = pt3d.normalize( Mc(-0.5*sin(t2/2),t2,0.5*cos(t2/2)) )
    table.insert(L1,{P1, P1 + len*nv1})
    table.insert(L2,{P2, P2 - len*nv2})
end
g:Dscene3d( g:addWall(wall),
    g:addFacet(s, {color="Gold", twoside=false, opacity=0.8, edge=true, edgecolor="gray"}),
    g:addPolyline(L1, {width=8, color="blue", arrows=1, arrowscale=0.5}),
    g:addPolyline(L2, {width=8, color="ForestGreen", arrows=1, arrowscale=0.5})
    )
g:Show()
\end{luadraw}
\end{document}
