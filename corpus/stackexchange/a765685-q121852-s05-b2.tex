\documentclass[tikz]{standalone}
\usepackage[3d]{luadraw}
\usepackage[svgnames]{xcolor}
%https://tex.stackexchange.com/questions/121852/how-can-i-draw-two-intersecting-spheres-with-highlighted-intersection
\begin{document}
\begin{luadraw}{name=sphere_intersection_wireframe}
local ld = luadraw
local pt3d = ld.pt3d
local M, Origin, vecI, vecJ, vecK = pt3d.M, pt3d.Origin, pt3d.vecI, pt3d.vecJ, pt3d.vecK
local g = ld.graph3d:new{ window={-5,5,-5,3}, viewdir={25,60} }
local O1, O2 = M(3,0,0), M(1,3,0.5)
local R1, R2 = 3, 2
local n = O2-O1
local t = (R1^2-R2^2)/(2*pt3d.abs2(O2-O1))
local J = (O1+O2)/2+t*(O2-O1)
local P = {J, n} -- radical plane of the two spheres
local r = math.sqrt(R2^2-pt3d.abs2(O2-J)) -- radius of the intersection circle
local H = J + r*pt3d.normalize( pt3d.prod(n,vecK) ) -- a point on the intersection circle
local S1, S2 = ld.sphere(O1,R1), ld.sphere(O2,R2)
S1 = g:Classifyfacet(S1) -- visible facets only
S2 = g:Classifyfacet(S2)
local _, S12 = ld.cutfacet(S1,P) -- cut the sphere with the plane P
local S21 = ld.cutfacet(S2,P)
local A, B = table.unpack( g:Sphere_tangency(O1,R1,P) ) -- tangency points of the intersection circle
local o1, r1, n1 = table.unpack( g:Sphere_outline(O1,R1).data ) -- Apparent circle of sphere S1
local o2, r2, n2 = table.unpack( g:Sphere_outline(O2,R2).data ) --Apparent circle of sphere S2

g:Dpolyline3d(S21,true,"orange")
g:Dcircle3d(J,r,n,"draw=none,pattern=north east lines, pattern color=Navy")
g:Dpolyline3d(S12,true,"cyan")
g:Dpath3d( {A,J,B,r,-1,n,"ca"}, "Navy, line width=0.8")
g:Dpath3d( {A,J,B,r,1,n,"ca"}, "Navy,dashed, line width=0.8")
g:Dpath3d({A,o1,B,r1,1,n1,"ca"}, "dashed,cyan")
g:Dpath3d({A,o2,B,r2,-1,n2,"ca"}, "dashed,orange")
g:Dlabel3d( "$P$",H,{pos="NE",node_options="Navy"}, 
    "$O_1$", O1, {pos="N",node_options="gray"}, "$O_2$", O2, {})
g:Ddots3d({O1,O2},"gray"); g:Ddots3d(H,"Navy")
g:Show()
\end{luadraw}
\end{document}
