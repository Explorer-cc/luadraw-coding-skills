\documentclass[12pt]{standalone}
\usepackage[svgnames]{xcolor}
\usepackage[3d]{luadraw}
\begin{document}
\begin{luadraw}{name=two_cylinders2}
local g = graph3d:new{size={12,12},window={-6,6,-3,3},viewdir={30,70}}
local R = 1.5
local C1 = cylinder(Origin,R,5*vecJ,45,true)
local C2 = cylinder(-5*vecI,R,5*vecI,45,true)
g:Dscene3d(
    g:addPoly(C1, {color="Chocolate",contrast=0.5}),
    g:addPoly(C2, {color="Chocolate",contrast=0.5})
    )
g:Show()
\end{luadraw} 
\end{document}
