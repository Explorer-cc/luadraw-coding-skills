\documentclass[border=5pt]{standalone}% compile with lualatex only
\usepackage[svgnames]{xcolor}
\usepackage[3d]{luadraw}%https://github.com/pfradin/luadraw
\usepackage{fourier-otf}
%https://tex.stackexchange.com/questions/123686/tikz-two-ellipses-in-different-planes
\begin{document}
\begin{luadraw}{name=two_ellipse3d}
local ld = luadraw
local pt3d, cpx = ld.pt3d, ld.cpx
local M = pt3d.M
local O, vI, vJ, vK = pt3d.Origin, pt3d.vecI, pt3d.vecJ, pt3d.vecK
local sin, cos, pi = math.sin, math.cos, math.pi

local g = ld.graph3d:new{ viewdir={"ortho",30,60}, window={-9,6,-6,8},
    size={12,12}, bbox=true }
local P1 = {M(0,-2,0), vK}
local P2 = {M(0,-2,1), vI+vK/2}
local A, u, v = ld.orthoframe(P2,vJ)
local E1 = ld.parametric3d( function(t) return O+4*cos(t)*vI+2*sin(t)*vJ end, -pi,pi,100)[1]
local E2 = ld.parametric3d( function(t) return A+3*cos(t)*u+4*sin(t)*v end, -pi,pi,100)[1]
g:Dscene3d(
    g:addPlane(P1,{rectangle={vJ,9,11}, color="red",opacity=0.2, edge=true, edgecolor="red"}),
    g:addPlane(P2,{rectangle={vJ,9,11}, color="blue",opacity=0.2, edge=true, edgecolor="blue"}),
    g:addFacet(E1, {color="red", opacity=0.5, edge=true, edgecolor="red"}),
    g:addFacet(E2, {color="blue", opacity=0.5, edge=true, edgecolor="blue"}),
    g:addDots({O,A}),
    g:addPolyline({{O-4*vI,O+4*vI},{O-2*vJ,O+2*vJ},{A-3*u,A+3*u},{A-4*v,A+4*v}},{arrows=1})
)
g:Dline3d( ld.interPP(P1,P2) ,"ForestGreen")
g:Dlabel3d("$E_2$",A+4*v,{pos='N',dir={u,v}}, "$E_1$",O+4*vI,{pos='S',dir={vJ,-vI}})
g:Show()
\end{luadraw}
\end{document}
