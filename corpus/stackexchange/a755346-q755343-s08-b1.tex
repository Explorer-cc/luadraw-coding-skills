\documentclass[border=5pt]{standalone}% compile with lualatex only
\usepackage[svgnames]{xcolor}
\usepackage[3d]{luadraw}%https://github.com/pfradin/luadraw
\usepackage{fourier-otf}
%https://tex.stackexchange.com/questions/755343/how-can-i-draw-grid-of-cylinders
\begin{document}
\begin{luadraw}{name=grid_of_cylinders}
local g = graph3d:new{window={-6,6,-4,9},bg="lightgray" }
Hiddenlinestyle = "noline"
local r = 0.5
local color = "orange"
g:Linewidth(2); g:Linecolor(color)
local Dcrossing = function(center)
    local Dcorner = function(A) -- A=M(+/-r, +/-r,+/-r)
        local c1, c2, c3 = px(A),py(A),pz(A)
        local a, b, c = pxz(A), pyz(A), pxy(A)
        g:Dpath3d( {a,c1,c,r,1,"ca",c2,b,r,1,"ca",c3,a,r,1,"ca"},"fill="..color.."!30")
    end
    g:Savematrix(); g:Shift3d(center)
    for x = -1,1,2 do
        for y = -1,1,2 do
            for z = -1,1,2 do
                local A = M(x*r,y*r,z*r)
                if pt3d.dot(A,g.Normal) > 0 then Dcorner(A) end -- only if A is visible
            end
        end
    end
    g:Restorematrix()
end
local Shift = function(L,u)
    local f = function(A)
        if isPoint3d(A) then return A+u else return A end
    end
    return ftransform3d(L,f)
end
local grid_base = {} -- list of {center, "type"}
for y = -4, 4  do
    local A = y*2*r*vecJ
    if y%3 == 0 then
        table.insert(grid_base, {A,"c"}) -- crossing
        table.insert(grid_base, {A+2*r*vecI,"x"}) -- tube along the Ox axis
        table.insert(grid_base, {A-2*r*vecI,"x"}) -- tube along the Ox axis
        table.insert(grid_base, {A+2*r*vecK,"z"}) -- tube along the Oz axis
        table.insert(grid_base, {A-2*r*vecK,"z"}) -- tube along the Oz axis
    else
        table.insert(grid_base, {A,"y"}) -- tube along the Oy axis
    end
end
local k = 6*r
local grid = concat(grid_base, Shift(grid_base,k*vecI), Shift(grid_base,-k*vecI)) -- layer in the plane z=0
grid = concat(grid, Shift(grid,k*vecK), Shift(grid,2*k*vecK))  -- adding layers in the z=3 and z=6 planes
table.sort(grid, function(e1,e2) return pt3d.dot(e1[1],g.Normal) < pt3d.dot(e2[1],g.Normal) end) -- sorting of elements according to distance
-- drawing
for _,elt in ipairs(grid) do
    local A, mode = table.unpack(elt)
    local U = nil
    if mode == "c" then Dcrossing(A)
    elseif mode == "x" then U = vecI
    elseif mode == "y" then U = vecJ
    elseif mode == "z" then U = vecK
    end
    if U ~= nil then g:Dcylinder(A-r*U,r,A+r*U, {color=color}) end
end 
g:Show()
\end{luadraw}
\end{document}
