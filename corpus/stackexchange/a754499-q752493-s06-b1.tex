\documentclass[border=5pt]{standalone}%compile with lualatex only
\usepackage[svgnames]{xcolor}
\usepackage{luadraw}%https://github.com/pfradin/luadraw/
%https://tex.stackexchange.com/questions/752493/tikz-ellipse-with-arrows
\begin{document}
\begin{luadraw}{name=curvilinear_param}
local g = graph:new{bbox=false,size={10,10}}
local i = cpx.I; g:Linewidth(8)

function Decorated(L,nb,arrow,close)
-- L is a polygonal line
-- nb is the number of arrows
-- arrow is the decoration
-- close = true if L must be closed before drawing
    close = close or false
    local f = curvilinear_param(L,close) -- f is a function, it's a curvilinear parametrization of L ( L={f(t); t in [0;1]} )
    local t = linspace(0,1,3*nb+1) -- 3*n+1 points equally distributed in [0;1]
    for k = 0,nb-1 do
        g:Dparametric(f,{t={t[3*k+1],t[3*k+3]},nbdots=10,nbdiv=2,draw_options=arrow}) -- we draw the curved arcs
    end
end

-- first example
local L = ellipse(0,4,2) 
g:Shift(2*i)
g:Linecolor("Crimson")
Decorated(L,8,"-stealth") -- or "stealth-" to change the direction of the arrows

-- another example
L = cartesian(math.sin,-5,5)[1] -- L is a polygonal line
insert(L, {5-2*i, -5-2*i}) -- we add two points at the end
g:Shift(-4*i)
g:Linecolor("ForestGreen")
Decorated(L,16,"-stealth",true)
g:Show()
\end{luadraw}
\end{document}
