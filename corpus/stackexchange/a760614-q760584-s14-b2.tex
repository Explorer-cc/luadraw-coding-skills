\documentclass[border=5pt]{standalone}% compile with lualatex only
\usepackage[svgnames]{xcolor}
\usepackage[3d]{luadraw}%https://github.com/pfradin/luadraw
\usepackage{fourier-otf}
%https://tex.stackexchange.com/questions/760584/drawing-a-topological-surface-of-genus-three
\begin{document}
\begin{luadraw}{name=surface_of_genus_three}
local g = graph3d:new{viewdir=perspective("central"), window={-4.5,4.5,-4,4.5}, size={12,12}, bg="lightgray"}
local R, r, cos, sin, pi = 3, 1, math.cos, math.sin, math.pi
-- to make a generic connection
local c = 0.3
local a, b, c1, c2 = 2.25*c, Z(c,2*c), c, Z(c,c)
local chem = parametric( function(t) return (1-t)^3*a+3*(1-t)^2*t*c1+3*t^2*(1-t)*c2+t^3*b end, 0,1,6,false,0)[1]
table.insert(chem,1,3*c)
local L = map(function(z) z=toComplex(z); return M(z.re,0,z.im) end, chem )
local connection = shift3d( rotline(L,{Origin, vecK}, 360,0), M(R,0,r) )
local f = function(A)
        local B = M(A.x,A.y,0)
        local I = R*pt3d.normalize(B)
        if A.z < r+c/8 then 
            return I + r*pt3d.normalize(A-I)
        elseif A.z < r+c/4 then 
            return I + (r+0.05)*pt3d.normalize(A-I)
        elseif A.z < r+c/2 then 
            return I + (r+0.085)*pt3d.normalize(A-I)
        else return A
        end
end
connection = ftransform3d(connection,f)
-- angles and axe for connections
local theta1, theta2, axe = 45, 135, {Origin, vecK}
-- make the torus
local h = function(t) return M(0,R+r*cos(t),r*sin(t)) end
local T = rotcurve(h,-pi,pi,{Origin,vecK}, 360,0, {grid={30,45}}) --torus
-- make a generic handle
local handle = function(A, B)
    local L = arc3d(B,(A+B)/2,A,pt3d.abs(B-A)/2,1,pt3d.prod(B-A,vecK))
    return line2tube(L,c,{nbfacet=25})
end
-- drawing
local nb = #palGasFlame
local pal1 = map( function(k) return palGasFlame[k] end, range(1,nb//2)) 
local pal2 = map( function(k) return palGasFlame[k] end, range(nb//2,nb)) 
g:Dfacet(T, {usepalette={pal1,"z"}, backcull=true, edgecolor="LightBlue!60", edgewidth=1}) 
g:Dfacet(concat(
            rotate3d(connection,theta1,axe), 
            rotate3d(connection,theta2,axe), 
            rotate3d(connection,-theta1,axe), 
            rotate3d(connection,-theta2,axe),
            handle(Mc(R,theta1*deg,r+2*c), Mc(R,theta2*deg,r+2*c)),
            handle(Mc(R,-theta1*deg,r+2*c), Mc(R,-theta2*deg,r+2*c)) ),
            {usepalette={pal2,"z"}, backcull=true, edgecolor="LightBlue!60", edgewidth=1} ) 
g:Show()
\end{luadraw}
\end{document}
