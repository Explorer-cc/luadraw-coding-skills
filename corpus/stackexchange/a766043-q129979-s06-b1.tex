\documentclass{standalone}
\usepackage[3d]{luadraw}
\usepackage[svgnames]{xcolor}
\usepackage{fourier-otf}
%https://tex.stackexchange.com/questions/129979/how-to-draw-spherical-triangles
\begin{document}
\begin{luadraw}{name=spherical_triangles}
require 'luadraw_spherical'
require 'luadraw_decorations'
local ld = luadraw
local pt3d = ld.pt3d
local Origin = pt3d.Origin
local g = ld.graph3d:new{ window={-4.5,4.5,-4.5,4.5}, viewdir={30,65}, size={10,10} }
ld.Hiddenlines = true
local Ct, R = Origin, 4
local function Add_angle(B,A,C,r,sens,name) -- add a spherical angle
    local options = {label=name, dist=0.5, pos="center", arc_options="line width=1"}
    local toSphere = function(N) return Ct+R*pt3d.normalize(N-Ct) end
    A = toSphere(A); B = toSphere(B); C = toSphere(C); 
    local C1 = toSphere( A+r*pt3d.normalize(C-A) )
    local B1 = toSphere( A+r*pt3d.normalize(B-A) )
    g:Dpath3d( {A,Ct,B1,R,1,"ca",A,C1,r,1,"ca",Ct,A,R,1,"ca"}, "draw=none,fill=green!30,fill opacity=0.6")
    g:Darc3d(B1,A,C1,r,sens,options)
end
g:Define_sphere{center=Ct, radius=R, edgestyle = "noline"}
local sM = ld.sM
local A,B,C = sM(30,25), sM(-20,80), sM(80,85)

g:DSfacet({A,B,C}, {fill="blue!50",color="blue", fillopacity=0.5, width=10})
g:DSpolyline({{Ct, A}, {Ct, B}, {Ct, C}}, {width=8})
g:DSdots(Ct)
g:DSlabel("$O$",Ct,{pos="NE"})
g:Dspherical()
Add_angle(B,A,C,1,1, "$\\alpha$")
Add_angle(A,B,C,1,1,"$\\beta$")
Add_angle(A,C,B,1,1,"$\\gamma$")
g:Dballdots3d({A,B,C},"black")
g:Dlabel3d(
    "$B$",B,{pos="SW"},
    "$C$",C,{pos="SE"},
    "$A$",A,{pos="N",dist=0.1}
)
g:Show()
\end{luadraw}
\end{document}
