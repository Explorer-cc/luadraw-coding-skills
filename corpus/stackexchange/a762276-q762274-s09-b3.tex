\documentclass[border=5pt]{standalone}
\usepackage[svgnames]{xcolor}
\usepackage[3d]{luadraw}
\begin{document}
\begin{luadraw}{name=sphere_clip3d_fixed3}
local R, I = 3, 1.5*vecK
local r = math.sqrt(9-1.5^2)
local g = graph3d:new{window={-R-2, R+1, -R-1, R}, viewdir={110, 60}, size={10, 10}}
Hiddenlinestyle = "dashed"
local vector = 2*vecK + 5*vecJ 
require "luadraw_povray"
g:Pov_new()
g:Pov_sphere(Origin, R, {clipplane={I,-vecK}, color=Orange})
g:Pov_circle(I,r,vecK); g:Pov_circle(I+vector,r,vecK); 
g:Pov_sphere(Origin+vector, R, {clipplane={I+vector,vecK}, color=Orange})
g:Pov_exec(); g:Pov_show()
g:Dcircle3d( I,r,vecK, Hiddenlinestyle); g:Dcircle3d( I+vector,r,vecK, Hiddenlinestyle)
g:Show()
\end{luadraw}
\end{document}
