\documentclass[12pt]{standalone}
\usepackage[svgnames]{xcolor}
\usepackage[3d]{luadraw}
\usepackage{fouriernc}
\begin{document}
%https://tex.stackexchange.com/questions/522965/how-can-i-repair-dimension-too-large-in-this-code?noredirect=1&lq=1
\begin{luadraw}{name=SSStriangle_13_10_2025}
local ab,bc,ac =135,352,377
local T =  sss_triangle3d(ab,bc,ac)
local x1,x2,y1,y2,z1,z2 = getbounds3d(T)
local g = graph3d:new{ window3d={x1,x2,y1,y2,z1,z2}, adjust2d=true,viewdir={300,60},size={12,12}}

local A,B,C = table.unpack(T)
local I,r,n = incircle3d(A,B,C)
g:Dfacet({A,B,C},{color="Orange",edgecolor="Blue",edgewidth = 6})
g:Dcircle3d(I,r,n) 
g:Ddots3d({A,B,C,I})
g:Dlabel3d("$A$",A,{pos="S"}, "$B$",B,{},"$C$",C,{pos="N"},"$I$",I,{pos="S"})
g:Show()
\end{luadraw}
\end{document}
