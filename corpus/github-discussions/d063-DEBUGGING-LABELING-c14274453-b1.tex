\documentclass{standalone}
\usepackage{xcolor}
\colorlet{myteal}{teal!50}
\colorlet{mymagenta}{magenta!50}
\colorlet{mylime}{lime!50}
\usepackage[3d]{luadraw}
\begin{document}
    \begin{luadraw}{name=icosaedre}
    local g = graph3d:new{ 
        window={-2.25,2.25,-2.25,2},
        viewdir={5,108}, 
        size={10,10}, 
        margin={0,0,0,0}
    }
    g:Linejoin("round")
    local phi = (1+math.sqrt(5))/2   -- golden ratio
    -- https://en.wikipedia.org/wiki/Icosahedron#Cartesian_coordinates
    local prefix = {"A","B","C","D"}    -- for the names
    local A1, B1, C1, D1 = M(phi,-1,0), M(phi,1,0), M(-phi,1,0), M(-phi,-1,0)   -- plane z=0
    local A2, B2, C2, D2 = M(0,phi,1), M(0,phi,-1), M(0,-phi,-1), M(0,-phi,1)   -- plane x=0
    local A3, B3, C3, D3 = M(1,0,phi), M(-1,0,phi), M(-1,0,-phi), M(1,0,-phi)   -- plane y=0
    local ico1 = {
        {A1,B1,A3},{B3,A3,A2},{A3,B3,D2},{B1,A2,A3},{A1,A3,D2}
    }
    local ico2 = {
        {B1,A1,D3},{C1,D1,B3},{B2,A2,B1}, {A2,B2,C1}, {D2,C2,A1}, {C2,D2,D1}, {A2,C1,B3}, {D1,D2,B3},{B2,B1,D3}, {A1,C2,D3}
    }
    local ico3 = {
        {C2,D1,C3},{D3,C3,B2},{C3,D3,C2},{B2,C3,C1},{D1,C1,C3}
    }
    local ico = facet2poly(concat(ico1, ico2, ico3)) -- facet to polyhedron conversion
    g:Dscene3d( 
        g:addFacet(ico1,{color="myteal"}),
        g:addFacet(ico2,{color="mymagenta"}),
        g:addFacet(ico3,{color="mylime"}),
        g:addPolyline(facetedges(ico),{width=12,hidden=true,hiddenstyle="dashed"})
        --g:addDots(ico.vertices, {color="black",scale=1.2}) -- this will hidden invisble dots...
    )
    local labels = {}
    for k,X in ipairs(ico.vertices) do
        local j = (k-1)//3+1 -- index in prefix table
        local p = (k%3)+1 -- index in {1,2,3}
        insert(labels,{"$"..prefix[j].."_"..p.."$",X,{pos="S",dist=0.1}})
    end
    g:Dballdots3d(ico.vertices,black, 1.2)
    g:Dlabel3d(table.unpack(labels))
    g:Show() 
    \end{luadraw}
\end{document}
