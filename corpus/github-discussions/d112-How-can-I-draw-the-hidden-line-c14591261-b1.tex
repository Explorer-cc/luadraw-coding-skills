\begin{luadraw}{name=sphere_cone}
local g = graph3d:new{window={-5,6,-8,5},viewdir={20,70},size={10,10}}
Hiddenlines = true; Hiddenlinestyle = "dashed";g:Linewidth(4);
local R,r=4,3.5
local h = math.sqrt(R^2 - r^2)
local a =math.asin(r/R)
local H=r*math.tan(a)
local d = h + H
local O,T = Origin, M(0,0,-d)
local S = poly2facet( sphere(Origin,R,35,35) )
S = g:Classifyfacet( cutfacet(S, {M(0,0,H-d),vecK}) )
g:Dcone(M(0,0,-d),H*vecK,r,{color="DarkBlue"})
g:Dpolyline3d( border(S), "ball color=orange, line width=0.6pt")
g:Dcircle3d(M(0,0,H-d),r,vecK,"dashed")
g:Dcircle3d(M(0,0,0),R,g.Normal,"dashed")
g:Ddots3d({O,T})
g:Dpolyline3d( {T,O}, "dashed")
g:Dlabel3d("$O$",O,{pos="N"}, "$T$",T,{pos="S"})
g:Show()
\end{luadraw}

\begin{luadraw}{name=sphere_cone2}
local g = graph3d:new{window={-5,6,-8,5},viewdir={20,70},size={10,10}}
Hiddenlines = true; Hiddenlinestyle = "dashed";g:Linewidth(4);
require 'luadraw_spherical'
local R,r=4,3.5
local h = math.sqrt(R^2 - r^2)
local a =math.asin(r/R)
local H=r*math.tan(a)
local d = h + H
local O,T = Origin, M(0,0,-d)
local C = cone(M(0,0,-d),H*vecK,r,100,true) 
local Cv, Ch = g:Classifyfacet(C)

g:Define_sphere({radius=R, edgecolor="black", edgewidth=6})
g:DScircle({M(0,0,H-d),vecK})
g:Dspherical()
g:Dpolyline3d( border(Cv), "left color=DarkBlue!50, right color=DarkBlue, middle color=DarkBlue!10") --cone
g:Dcircle3d(M(0,0,0),R,g.Normal,"dashed")  -- border of sphere
g:Ddots3d({O,T})
g:Dpolyline3d( {T,O}, "dashed")
g:Dlabel3d("$O$",O,{pos="N"}, "$T$",T,{pos="S"})
g:Show()
\end{luadraw}
