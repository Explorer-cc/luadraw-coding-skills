\documentclass{standalone}
\usepackage[svgnames]{xcolor}
\usepackage[3d]{luadraw}
%https://tex.stackexchange.com/questions/734005/tikz-3d-graphic-background-foreground-intersections
\begin{document}
 \begin{luadraw}{name=answer}
local g = graph3d:new{window={-2,8,-3,7},viewdir={20,70},size={10,10}}
Hiddenlinestyle = "dashed"; Hiddenlines = true;g:Linewidth(6);
local a, b, h,k = 4,8,5,8
local P = parallelep(Origin, a*vecI,b*vecJ,h*vecK)
local A,B,C,D,E,F,G,H = table.unpack(P.vertices)
local F1 = getfacet(P,{1,3,4,6}) -- remove back and top faces of P
local K, L = M(a/2,b/4,k),M(a/2,3*b/4,k) 
local P = facet2poly( concat(F1,{{K,E,F},{K,F,G,L},{L,G,H},{K,L,H,E}}) )
local A = g:Edges(P)
g:Dpolyline3d(A.hidden,"dotted")
g:Dpolyline3d({{F,G},{C,G,L}},"line width=3pt,draw=white,double=black,double distance=0.6pt")
g:Dpolyline3d(A.visible)
g:Show()
\end{luadraw}
\end{document}
