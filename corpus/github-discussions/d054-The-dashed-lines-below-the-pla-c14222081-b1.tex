%https://tex.stackexchange.com/questions/728955/graphing-the-intersection-of-2-planes-on-overleaf
\begin{luadraw}{name=intersection_two_planes}
local g = graph3d:new{ window3d={-10,10,-10,10,-10,10},window={-10,10,-12,11},viewdir={30,60},size={10,10}}
g:Linejoin("round"); g:Linewidth(4); Hiddenlinestyle = "dashed"
local P = planeEqn(4,-1,3,-4)
local Q = planeEqn(2,-3,4,3) 
local I = interPP(P, Q) 
Hiddenlines = true
g:Dscene3d(
    g:addPlane(P, {edge=true, color="blue",scale=0.65}),
    g:addPlane(Q, {edge=true, color="ForestGreen",scale=0.65}), 
    g:addLine(I, {color="red",width=12,scale=0.75}),
    g:addAxes(Origin, {arrows=1, color="Gold", width=8})
    )
g:Show()
\end{luadraw} 
\end{document}
