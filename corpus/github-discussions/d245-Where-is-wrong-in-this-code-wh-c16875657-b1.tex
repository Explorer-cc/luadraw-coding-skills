\documentclass[border=2 mm]{standalone}
\usepackage[svgnames]{xcolor}
\usepackage[3d]{luadraw}
\usepackage{fouriernc}
\begin{document}
\begin{luadraw}{name=fill}
local ld = luadraw
local cpx = ld.cpx
local sqrt = math.sqrt
local Z, i = cpx.Z, cpx.I
local xmin, xmax, ymin, ymax = -1, 5, -2, 4
local g = ld.graph:new{window={xmin, xmax, ymin, ymax}}  
g:Daxes({0,1,1}, {grid=true, gridstyle = "dashed",arrows="-Stealth", legend={"$x$","$y$"}})
g:Linewidth(8)
g:Linecap("round")
local f = function(x) return 4*x-11  end
local h = function(x) return sqrt(x^2-4) end
local k = function(x) return -sqrt(x^2-4) end
g:Dinequalities({f,'>', h,'<'}, {view={2,4,0,5},draw_options="draw=none,fill=green,fill opacity=0.6"})
g:Dinequalities({f,'>', k,'>'}, {view={2,3,-2,0},draw_options="draw=none,fill=green,fill opacity=0.6"})
g:Dcartesian(f, {draw_options="blue"});
g:Dcartesian(h, {x={2,5},draw_options="violet"});
g:Dcartesian(k, {x={2,5},draw_options="violet"});  
g:Show()
\end{luadraw}
\end{document}
