\documentclass[tikz]{standalone}
\usepackage{fourier-otf}
\usepackage[svgnames]{xcolor}
\usepackage[3d]{luadraw}

\begin{document}

\begin{luadraw}{name=saddle_surface}
local ld = luadraw
local pt3d = ld.pt3d
local M = pt3d.M
local g = ld.graph3d:new{size={10,10},window3d={-2,2,-2,2,-2,2},window={-3,3,-2.5,1.5}, viewdir={-115,60}}
--window={-2,1.5,-3,1.5}
local f = function(x,y) return x^2/4-y^2/3 end

local dot = function(x,y) return M(x,y,f(x,y)) end

local visible = function(A) -- A in on the surface
    local x, y, h = A.x, A.y, 1e-6
    local B, C = dot(x+h,y), dot(x+h,y+h)
    local N = pt3d.prod(B-A,C-A)
    return g:Cosine_incidence(N, A) > 0 -- N is towards observer
end

local A, B, C = dot(0,1.8), dot(-1.4,-1.65), dot(1.4,-1.65) -- three points on the surface
local fAB = function(t)
    local x = A.x + t*(B.x-A.x)
    local y = A.y + t*(B.y-A.y)
    return M(x,y,f(x,y))
end

local fCA = function(t)
    local x = C.x + t*(A.x-C.x)
    local y = C.y + t*(A.y-C.y)
    return M(x,y,f(x,y))
end

local fBC = function(t)
    local x = B.x + t*(C.x-B.x)
    local y = B.y + t*(C.y-B.y)
    return M(x,y,f(x,y))
end

local tri = function(t)
    if t < 1 then return fAB(t)
    elseif t < 2 then return fBC(t-1)
    else return fCA(t-2)
    end
end
local S = ld.cartesian3d(f,-2,2,-2,2,{30,30})
local L = ld.parametric3d(tri,0,3)
local Lv, Lh = ld.split_points_by_visibility(L, visible)

g:Dfacet(S,{color="black",mode=ld.mShaded,edge=true,edgecolor="white"})
g:Dpolyline3d(Lv, "red,line width=1.5")
g:Dpolyline3d(Lh, "red,line width=1.5,dashed")
g:Show()
\end{luadraw}
\end{document}
