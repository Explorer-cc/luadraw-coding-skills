\documentclass[12pt]{standalone}
\usepackage[svgnames]{xcolor}
\usepackage[3d]{luadraw}
\usepackage{fouriernc}
\begin{document}

\begin{luadraw}{name=AMC}
local g = graph3d:new{window={-2.5,2.5,-0.5,3.5},viewdir={-40,60},size={10,10}}
g:Linewidth(4); Hiddenlines = true; Hiddenlinestyle = "dashed"
local h = 2.5
local base = {M(-1.5, 1.5, h), M(1.5, 1.5, h), M(1.5, -1.5, h), M(-1.5, -1.5, h)} -- base of the pyramid
local apex = -h*vecK/2 -- apex of the pyramid
local P1 = scale3d( reverse(base), 2/3, apex) -- water
g:Dscene3d(
    g:addFacet(P1, {color="cyan"}),
    g:addPoly(truncated_pyramid(base,apex,h,true), {color="Orange", edge=true})
)
g:Show()
\end{luadraw}
\end{document}
