\begin{luadraw}{name=con_out_cube}
local g = graph3d:new{window={-6,1.5,-2,5},viewdir={80,70},size={10,10}}
g:Linejoin("round"); Hiddenlinestyle = "dashed"
local a = 4
local P = parallelep(Origin, a*vecI,a*vecJ,a*vecK)
local A,B,C,D,E,F,G,H = table.unpack(P.vertices)
g:Dpoly(P,{mode=3,color="LightBlue"})
local I=(D+B+G)/3
local R = pt3d.abs(I-B)
g:Dcone(C,I-C,R,{color="orange"})
g:Linestyle(Hiddenlinestyle) 
g:Dpoly(P,{mode=0,edgewidth=1})  -- after the cone
g:Dpolyline3d( {{C,I}, {D,B,G,D}})
 
g:Dlabel3d("$A$",A,{pos="S"}, "$B$",B,{}, "$C$",C,{}, "$D$",D,{},"$E$",E,{pos="N"}, "$F$",F,{}, "$G$",G,{}, "$H$",H,{}, "$I$",I,{})
 g:Ddots3d({A,B,C,D,E,F,G,H,I})
g:Show()
\end{luadraw}
