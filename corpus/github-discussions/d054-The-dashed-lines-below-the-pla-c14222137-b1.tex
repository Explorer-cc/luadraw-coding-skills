\begin{luadraw}{name=pyramid2_12_08_2025}
local g = graph3d:new{window3d={-8,8,-8,8,-8,8}, window={-5,5,-2,6.5}, viewdir={280,70}, size={10,10}}
g:Linejoin("round"); Hiddenlines = true; g:Linewidth(5);Hiddenlinestyle = "dashed"
local a=  4
local h = a*math.sqrt(2)
  local  A, B, C, D, S = Origin, M(a,0,0), M(a,a,0), M(0,a,0), M(0,0,h)
local O, M,I = (A+C)/2, (S+A)/2,(S+D)/2
local H = dproj3d(A,{S, S-O})
local E = dproj3d(A,{S, S-D})
local K = B-A+E
local d2 ={B,B-M}
local d3 ={C,C-I}
local N = interDP(d2,{S,pt3d.prod(S-C,S-D)})
local d1 =interPP({S,pt3d.prod(S-A,B-A)},{S,pt3d.prod(S-C,S-D)})
local P = pyramid( {A,B,C,D}, S)
g:Dscene3d(
g:addLine(d1, {color="red",width=5,scale=1.5}),
g:addPoly(P,{edge=true,color="LightBlue",edgewidth=5, edgecolor="Blue"}),
g:addPolyline({{S,O},{A,C},{B,D},{A,E,K,B},{A,H},{N,K},{M,I},{C,N},{B,N},{C,K}}),
g:addAngle(A,H,O),
g:addAngle(A,E,D)
)
--g:Dangle3d(A,E,S,0.15)
g:Ddots3d({O,S,A,B,C,D,H,E,K,M,I,N})
g:Dlabel3d( "$O$",O,{pos="S"}, "$S$",S,{pos="N"}, "$A$",A,{pos="S"}, "$B$",B,{pos="S"}, "$C$",C,{pos="E"}, "$D$",D,{pos="E"}, "$H$",H,{pos="E"}, "$E$",E,{pos="E"},"$K$",K,{pos="E"},"$M$",M,{pos="W"},"$I$",I,{pos="E"},"$N$",N,{pos="N"})
g:Show()
\end{luadraw}
\end{document}
