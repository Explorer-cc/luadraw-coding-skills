\documentclass[border=5pt]{standalone}
\usepackage{xcharter-otf}
\usepackage{luadraw}

\begin{document}

\begin{luadraw}{name=curve_tangent}
local ld = luadraw
local cpx = ld.cpx
local i, Z = cpx.I, cpx.Z
local g = ld.graph:new{window = {-0.7, 5.8, -0.35, 4.35}, size={12,12}}
clipline = ld.clipline -- to patch a bug in 'luadraw_lines.lua' file

local function f(x)
    return 3+0.125*x*(x+1)*(x-4.5)
end

local function df(x)
    local h = 1e-6
    return (f(x+h)-f(x-h))/(2*h)
end

local label_x = function(...)
    local seg, labels, p= {}, {}
    for k,v in ipairs{...} do
        if k%2 == 1 then 
            p = v
            table.insert(seg, {p, p.re})
        else
            ld.insert(labels, {v, p.re-0.15*i, {}})
        end
    end
    g:Dpolyline(seg, "dashed, thick")
    g:Dlabel( table.unpack(labels) )
end

local L = ld.cartesian(f,-1,5.5)
local x0 = 4 -- abscissa of c
local t0 = ld.solve(df,-1,1)[1]
local z1 = Z(t0,f(t0)) -- horizontal tangent
local c = ld.interDL({x0,i}, L)[1]
local z2 = ld.tangent_from(c, function(t) return Z(t,f(t)) end, -1,5)[1]
local z3 = ld.tangent_from(Z(x0,0), function(t) return Z(t,f(t)) end, -1,3)
table.sort(z3, function(a,b) return a.re < b.re end)
local e = z3[2]
z3 = z3[1]
local a = ld.interDL({z1,1}, L)[1]
local b = ld.interDL({z3,x0-z3}, L)[1]
-- drawing
g:Dpolyline({{-0.5, 5.65},{-0.2*i, 4.2*i}}, "-Stealth, very thick")
g:Dpolyline(L, "line width=0.8pt")
g:Linestyle("dashed")
g:Dline(z1,z1+1); g:Dline(z2,c); g:Dline(z3,x0); g:Dline(e,x0)
label_x(e, "$e$", c,"$c$", a, "$a$", b, "$b$") 
g:Show()
\end{luadraw}
\end{document}
