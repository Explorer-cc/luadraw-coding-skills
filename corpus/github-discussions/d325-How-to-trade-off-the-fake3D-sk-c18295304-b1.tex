\documentclass{standalone}
\usepackage[3d]{luadraw}
\usepackage[svgnames]{xcolor}
\usepackage{fourier-otf}
\begin{document}
\begin{luadraw}{name=hemisphere_tetra}
local ld = luadraw
local pt3d, cpx = ld.pt3d, ld.cpx
local M, Mc, Ms = pt3d.M, pt3d.Mc, pt3d.Ms
local Origin, vecI, vecJ, vecK = pt3d.Origin, pt3d.vecI, pt3d.vecJ, pt3d.vecK
local g = ld.graph3d:new{
    window = {-3,3,-3,2}, size = {12,12},
    viewdir = {30,70} -- should be fixed
}
ld.Hiddenlines = false; ld.Hiddenlinestyle= "dashed"
require 'luadraw_spherical'
local A,B,C = M(3,-1,0), M(0,3,-0.5), M(-1.5, -1.5,-1)
local P = ld.interSSS( {(A+B)/2, pt3d.abs(A-B)/2}, {(A+C)/2, pt3d.abs(A-C)/2}, {(C+B)/2, pt3d.abs(C-B)/2} )
local O, R = ld.circumsphere(A,B,C,P)
local sM = ld.sM
g:Define_sphere({center=O, radius=R, edgestyle="noline",show=false})
local T1, T2 = table.unpack( g:Sphere_tangency(O, R, {O, vecK}) )
local I, r, n = table.unpack( g:Sphere_outline(O,R).data )
local T
if pt3d.det(A-P, B-P, C-P) < 0 then 
    T = ld.tetra(P, B-P, A-P, C-P)
else 
    T = ld.tetra(P, A-P, B-P, C-P)
end
g:Dcircle3d(O,R,vecK, "draw=none, left color=lightgray, right color=white")
g:Dpoly(T, {color="gray!20!white",mode=ld.mShadedHidden})
g:Dangle3d(A,P,B); g:Dangle3d(A,P,C); g:Dangle3d(B,P,C)
g:DScircle({O,vecK}, {width=6})
g:DScircle( ld.plane(A,B,C), {hidden=true,width=6})
g:DSpolyline({{O,A},{O,B},{O,C}},{style="dashed",width=6})
g:DSdots({A,B,C,P,O})
g:Dspherical()
local S = {T1,I,T2,r,1,n, "ca", O, T1, R, -1, vecK, "ca"}
g:Beginclip( S)
    g:Dsphere(O, R, {color="lightgray", mode=ld.mBorder, opacity=0.3})
g:Endclip(true, "line width=0.6")
g:Dlabel3d(
    "$O$",O,{pos="N", dist=0.1}, "$A$", A, {pos="SW"},
    "$B$",B,{pos="SE"},
    "$C$",C,{pos="N"},
    "$P$",P,{pos="S"}
)
-- Verification
print(pt3d.angle(B-P,A-P)*ld.rad)
print(pt3d.angle(B-P,C-P)*ld.rad)
print(pt3d.angle(C-P,A-P)*ld.rad)
print("R=",R," OP=",pt3d.abs(P-O))
g:Show()
\end{luadraw}
\end{document}
