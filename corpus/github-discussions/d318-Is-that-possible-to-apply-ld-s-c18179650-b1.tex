\documentclass{standalone}
\usepackage[3d]{luadraw}
\usepackage[svgnames]{xcolor}
\begin{document}
    \begin{luadraw}{name=mobuis_border}
    local ld = luadraw
    local pt3d = ld.pt3d
    local Mc = pt3d.Mc
    local sin, cos, pi = math.sin, math.cos, math.pi
    local g = ld.graph3d:new{ window={-3,3,-2,2}, viewdir={-50,65}, size={12,12} }
    
    local r = function(u,v) return 2 + u * cos(v/2) end
    local z = function(u,v) return u*sin(v/2) end
    local mobius = function(u, v)
        return Mc(r(u,v),v,z(u,v))
    end

    local s1 = ld.cylindrical_surface(r,z,-0.7,0.7,-pi,pi,{5,101})
    local s2 = ld.cylindrical_surface(r,z,0.7,0.75,-2*pi,2*pi,{2,201})
    g:Dmixfacet(s1, {color = ld.LightGray,opacity = 0.75, twoside=false, mode=ld.mShadedOnly},
        s2, {color ="red"} )
    g:Show()
\end{luadraw}
\end{document}
