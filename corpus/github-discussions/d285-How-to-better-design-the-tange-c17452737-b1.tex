\documentclass[border=5pt]{standalone}
\usepackage{xcharter-otf}
\usepackage{luadraw}

\begin{document}

\begin{luadraw}{name=curve_tangent}
local ld = luadraw
local cpx = ld.cpx
local i, Z = cpx.I, cpx.Z
local g = ld.graph:new{window = {-0.7, 5.8, -0.35, 4.35}, size={12,12}}

g:Dpolyline({{-0.5, 5.65},{-0.2*i, 4.2*i}}, "-Stealth, very thick")

local label_x = function(p, label)
   g:Dpolyline({p, p.re}, "dashed, thick")
   g:Dlabel(label, p.re-0.15*i, {})
end

local dir = function(z)
   return cpx.arg(z)*ld.rad
end

local x0 = 3.75
local z1,z2,z3,z4,z5,z6,z7 = Z(-0.443,2.9), Z(-0.25,2.98), Z(-0.075,2.95), Z(0.1,2.85), Z(2, 0.8), Z(x0,1.4), Z(5,4)
local curve = {
z1, {0.0625,dir(z2-z1)}, -- {tension, angle in degrees}
z2, {0.25,0}, -- horizontal
z3, {0.125,dir(z6-z3), 0.25,dir(z6-z3)},
z4, {0.25,dir(x0-z4), 3,dir(x0-z4)},
z5, {3, dir(x0-z5)},
z6, {1.5, 55},
z7, {3,75}
}
local L = ld.path( ld.tcurve(curve) )
clipline = ld.clipline
local a = ld.interDL({z2,1}, L) 
local b = ld.interDL({z4,x0-z4}, L)
table.sort(a, function(z1,z2) return z1.re > z2.re end )
table.sort(b, function(z1,z2) return z1.re > z2.re end )
local c = z6

g:Dtcurve(curve,{showdots=false,draw_options="line width=0.8pt"})
g:Linestyle("dashed")
g:Dline(z4,x0); g:Dline(z5,x0); 
g:Dline(z2,z2+1); g:Dline({z3,z6-z3})
label_x(z5, "$e$"); label_x(a[1], "$a$"); label_x(b[1], "$b$"); label_x(c, "$c$") 
g:Show()
\end{luadraw}
\end{document}
