\documentclass{standalone}
\usepackage{luadraw}
\usepackage[svgnames]{xcolor}
\usepackage{fourier-otf}
\begin{document}
\begin{luadraw}{name=to_path}
local ld = luadraw
local cpx = ld.cpx
local g = ld.graph:new{window={-1,2.5,-1,3},size={10,10}}
local i, Z, Zp, deg = cpx.I, cpx.Z, cpx.Zp, ld.deg
local A, B = Z(0,2), Z(2,0)
local r = 0.3915*cpx.abs(B-A)
local c1, c2 = A+Zp(r,-25*deg), B+Zp(r,100*deg) -- control points
g:Dpath({A, c1, c2, B, "b"}, "line width=2, red")
-- verification
g:Writeln("\\draw "..g:Coord(A).." to[out=-25,in=100] "..g:Coord(B)..";")
g:Show()
\end{luadraw}
\end{document}
