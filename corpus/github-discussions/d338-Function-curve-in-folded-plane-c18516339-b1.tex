\documentclass{standalone}
\usepackage[3d]{luadraw}
\usepackage[svgnames]{xcolor}
\usepackage{fourier-otf}
\begin{document}
\begin{luadraw}{name=folded_sine_function}
    local ld = luadraw
    local pt3d, cpx = ld.pt3d, ld.cpx
    local M, Z, I = pt3d.M, cpx.Z, cpx.I
    local Origin, vecI, vecJ = pt3d.Origin, pt3d.vecI, pt3d.vecJ
    local sin, pi, sqrt = math.sin, math.pi, math.sqrt

    local g = ld.graph3d:new{ window = {-4, 4, -2, 4},viewdir = {-70, 70}, size = {10, 8} }
    local f = function(x) 
        return 1.5 * sin(pi/2 * (x + 5/3))
    end
    local x1, x2 = -3*pi/7, 5*pi/8
    local vecn = M(0, 0.5, sqrt(3)/2)
    g:Linewidth(8)
    g:BeginOnPlane({Origin,vecI,vecn}, {view = {-2.5,2.5,0,2.5}})
        g:Drectangle(-1.8, Z(2.4, 2.4)) -- <-- inside the view
        g:Dpolyline({{0,2.2*I}}, "-Stealth")
        g:Dcartesian(f,{x = {x1, x2},draw_options="magenta"})
    g:EndOnPlane()
    g:BeginOnPlane({Origin,vecI,vecJ}, {view = {-2.5, 2.5,-3.5,0}})
        g:Drectangle(-1.8, Z(2.4,-3.4)) -- <-- inside the view
        g:Dpolyline({0,-3*I})
        g:Dcartesian(f, {x = {x1, x2},draw_options="magenta"})
    g:EndOnPlane()
    g:Dpolyline3d({-1.8*vecI,2.4*vecI},"-Stealth")
    g:Show()
\end{luadraw}
\end{document}
